    95→function sh(cmd: string): string {
    96→  try {
    97→    return execSync(cmd, { cwd: ROOT, encoding: "utf8", timeout: 60_000, stdio: ["ignore", "pipe", "pipe"] }).toString().trim();
    98→  } catch (e) {
    99→    const err = e as { stderr?: string; stdout?: string };
   100→    throw new Error(`פקודה נכשלה: ${cmd}\n${(err.stderr || err.stdout || "").slice(-500)}`);
   101→  }
   102→}
   103→
   104→function gitOrNull(cmd: string): string | null {
   105→  try {
   106→    const out = execSync(cmd, { cwd: ROOT, encoding: "utf8", timeout: 30_000, stdio: ["ignore", "pipe", "pipe"] }).toString().trim();
   107→    return out || null;
   108→  } catch {
   109→    return null;
   110→  }
   111→}

   117→  try {
   118→    return JSON.parse(fs.readFileSync(LINE_PATH, "utf8")) as ZeroLine;

   120→    return { line: "zero-evm", rail: ZERO_RAIL_ID, chainId: 9369, account, createdAt: new Date().toISOString(), anchors: [] };

   124→function writeLine(line: ZeroLine): void {
   125→  const capped = { ...line, anchors: line.anchors.slice(-ANCHORS_RING) };

   127→  fs.writeFileSync(LINE_PATH, JSON.stringify(capped, null, 2) + "\n");
   128→}

   130→async function localLedgerState(): Promise<{ count: number; head: string }> {
   131→  const [last, count] = await Promise.all([
   132→    db.attestation.findFirst({ orderBy: { seq: "desc" } }),
   133→    db.attestation.count(),
   134→  ]);
   135→  return { count, head: last?.entryHash ?? "0x" + "0".repeat(64) };

   138→async function main(): Promise<void> {

   140→  const gitHead = gitOrNull("git rev-parse --short HEAD") ?? "unknown";

   146→  console.log(`[weave-anchor-zero] runner=${runner} head=${gitHead} rail=${ZERO_RAIL_ID}`);

   149→  if (PUSH_ONLY) {
   150→    if (!gitOrNull("git remote get-url origin")) {
   151→      throw new Error("אין remote origin — בלי ריפו אין עדות");

   164→    const pushOut = gitPushRetry(ROOT, BRANCH);
   165→    console.log(`[weave-anchor-zero] push-only: ${nFiles} files · ${pushOut.split("\n").slice(-1)[0] || "pushed"}`);

   173→      throw new Error("קונפליקט pull — הרץ ידנית: git pull --rebase ואז weave-anchor-zero");

   176→  }
   177→

   180→    const snap = parseLedgerSnapshot(fs.readFileSync(LEDGER_PATH, "utf8"));
   181→    const local = await localLedgerState();
   182→    if (snap.chainCount > local.count) {
   183→      const res = await importLedger(snap);

   186→  }
   187→
   188→  // 3. פתיחת seal — מפתח הרשת חותם את העוגן; זהות הענן חותמת את העדות
   189→  const seal = sealOpen();
   190→  const cloudAgent = await ensureCloudAgent(seal.cloudKeyHex);

   193→    `[weave-anchor-zero] seal פתוח · network=${seal.networkAddress} (חותם העוגן) · cloud=${seal.cloudAddress} (חותם העדות)`

   197→  //      נסוג כשהסנדבוקס חי (SANDBOX-ALIVE) — אבל הקווים
   198→  //      רצים אחרי כל פעימה בכל מצב. המאגר חי בספר
   199→  //      (כספת אדיטיבית) — ומדוד כאן, שעתית, ב-1-token chat
   200→  //      לכל מפתח. הסטטוס זורם לספר בייצוא שלהלן. כשל-רשת
   201→  //      או סביבה נטולת-מפתח — נרשם בכנות ולא מפיל: העשרה
   202→  //      לעולם לא חוסמת את הקו.

   204→  try {
   205→    const poolCount = await db.brainKey.count();
   206→    if (poolCount > 0) {
   207→      brainProbe = await brainProbeAll();
   208→      const live = brainProbe.filter((p) => p.live).length;
   209→      console.log(`[weave-anchor-zero] המוח: ${live}/${poolCount} מפתחות LIVE (אימות-אמת 1-token שעתי)`);

   212→    console.log(
   213→      `[weave-anchor-zero] המוח: נעול או לא-זמין — ${brainErr instanceof Error ? brainErr.message : String(brainErr)}`
   214→    );

   217→  // 4. הספר חייב להיות תקין לפני כל עדות
   218→  const verify = await verifyLedger();
   219→  if (!verify.valid) {
   220→    throw new Error(`הספר שבור (firstBreak=${verify.firstBreak ?? "?"}) — אין עדות על ספר שקרי`);
   221→  }
   222→  console.log(`[weave-anchor-zero] ledger תקין: ${verify.checked} אימותים · ${verify.sigOk}/${verify.checked} חתימות`);
   223→
   224→  // 4.5 התאוששות-עוגן — הטרנזקציה נחתה, הריצה המקורית אדמה לפני האימות
   225→  if (RECOVER_TXID) {
   226→    if (!/^0x[0-9a-fA-F]{64}$/.test(RECOVER_TXID)) {
   227→      throw new Error(`--recover: txid פגום — ${RECOVER_TXID.slice(0, 20)}…`);
   228→    }
   229→    const cpR = await db.checkpoint.findFirst({ orderBy: { index: "desc" } });
   230→    if (!cpR) throw new Error("אין checkpoint — אין מה לשחזר");
   231→    console.log(`[weave-anchor-zero] recovery: מאמת ${RECOVER_TXID.slice(0, 16)}… מול cp#${cpR.index}`);
   232→    const readback = await verifyZeroAnchorOnChain(RECOVER_TXID);
   233→    if (!readback.found || !readback.ok) {
   234→      throw new Error(`recovery: הטרנזקציה לא אומתה מהשרשרת — ${readback.verdict}`);
   235→    }
   236→    const p = readback.payload ?? {};
   237→    const pCheckpoint = Number(p.checkpoint ?? -1);
   238→    const pRoot = String(p.root ?? "");
   239→    if (pCheckpoint !== cpR.index || pRoot !== cpR.root) {
   240→      throw new Error(
   241→        `recovery: חוסר-התאמה מטען↔ספר — on-chain cp#${pCheckpoint}/${pRoot.slice(0, 14)}… מול ledger cp#${cpR.index}/${cpR.root.slice(0, 14)}…`
   242→      );
   243→    }
   244→    const originalAt = String(p.at ?? new Date().toISOString());
   245→    const originalCommit = String(p.commit ?? gitHead);
   246→    console.log(
   247→      `[weave-anchor-zero] recovery: מטען מאומת מהשרשרת (cp#${pCheckpoint}, root ${pRoot.slice(0, 16)}…) · block=${readback.block} · effGas=${readback.effectiveGasPrice}`
   248→    );
   249→
   250→    // ה-checkpoint מסומן ANCHORED (אם עוד לא)
   251→    if (cpR.anchorStatus !== "ANCHORED") {
   252→      await db.checkpoint.update({
   253→        where: { index: cpR.index },
   254→        data: { anchorStatus: "ANCHORED", anchorNet: ZERO_RAIL_ID, anchorTx: RECOVER_TXID },
   255→      });
   256→    }
   257→
   258→    const explorer = zeroExplorerLink(RECOVER_TXID);
   259→    const claim =
   260→      `קו ה-ZERO (Z Chain 9369, גז-אפס): checkpoint #${cpR.index} root ${cpR.root.slice(0, 18)}… עוגן כטרנזקציית EVM ` +
   261→      `אמיתית (recovery — העוגן נחת בריצה שאדמה במרוץ אינדקס לפני האימות; המטען אומת כעת מהשרשרת) — ` +
   262→      `עד EVM ציבורי בלתי-תלוי בגיטהאב, בעלות מדודה 0.000000.`;
   263→    const evidence = [
   264→      `zeroTxid=${RECOVER_TXID}`,
   265→      `recovery=true (original run reddened pre-verification; payload re-verified from chain)`,
   266→      `chainId=${ZERO_RAIL_ID} (Z Chain, Zero Network)`,
   267→      `block=${readback.block}`,
   268→      `gasUsed=${readback.gasUsed}`,
   269→      `effectiveGasPrice=${readback.effectiveGasPrice} (0 = הוכחת האפס בקבלה)`,
   270→      `costEth=0.000000`,
   271→      `account=${readback.from} (מפתח הרשת מה-seal)`,
   272→      `operation=self-addressed EOA · value=0 · data=0xa9059cbb+canonical`,
   273→      `checkpoint=${cpR.index} root=${cpR.root} attFrom=${cpR.attFrom} attTo=${cpR.attTo}`,
   274→      `originalAt=${originalAt}`,
   275→      `originalCommit=${originalCommit}`,
   276→      `readback=${readback.verdict}`,
   277→      `explorer=${explorer}`,
   278→      `runner=${runner} (recovery)`,
   279→      runUrl ? `run=${runUrl}` : null,
   280→    ]
   281→      .filter(Boolean)
   282→      .join(" · ");
   283→
   284→    const att = await appendAttestation({
   285→      agentId: cloudAgent.id,
   286→      claim,
   287→      claimType: "ANCHOR",
   288→      evidence,
   289→    });
   290→    console.log(`[weave-anchor-zero] recovery: אימות ANCHOR נכתב: seq=${att.seq}`);
   291→
   292→    const newCp = await createCheckpointNow();
   293→    if (newCp) {
   294→      console.log(`[weave-anchor-zero] recovery: checkpoint #${newCp.index} נוצר (מכסה att ${newCp.attFrom}..${newCp.attTo})`);
   295→    }
   296→
   297→    const lineR = readLine(seal.networkAddress);
   298→    if (!lineR.anchors.some((a) => a.txid === RECOVER_TXID)) {
   299→      lineR.anchors.push({
   300→        checkpoint: cpR.index,
   301→        root: cpR.root,
   302→        attFrom: cpR.attFrom,
   303→        attTo: cpR.attTo,
   304→        attestations: await db.attestation.count(),
   305→        at: originalAt,
   306→        commit: originalCommit,
   307→        account: readback.from ?? seal.networkAddress,
   308→        txid: RECOVER_TXID,
   309→        block: readback.block,
   310→        gasUsed: readback.gasUsed,
   311→        effectiveGasPrice: readback.effectiveGasPrice,
   312→        costEth: "0.000000",
   313→        confirmed: true,
   314→        attSeq: att.seq,
   315→        runner: runner,
   316→      });
   317→      writeLine(lineR);
   318→    }
   319→
   320→    const snapR = await exportLedger();
   321→    fs.mkdirSync(path.dirname(LEDGER_PATH), { recursive: true });
   322→    fs.writeFileSync(LEDGER_PATH, JSON.stringify(snapR, null, 2) + "\n");
   323→    fs.writeFileSync(
   324→      REPORT_PATH,
   325→      JSON.stringify(
   326→        {
   327→          at: new Date().toISOString(),
   328→          runner,
   329→          mode: "RECOVERED (anchor landed in a reddened run; attested after chain re-verification)",
   330→          recoveredTxid: RECOVER_TXID,
   331→          readback,
   332→          attSeq: att.seq,
   333→          nextCheckpoint: newCp ? { index: newCp.index, root: newCp.root } : null,
   334→          attestations: snapR.chainCount,
   335→          context: { runner, gitHead, repoSlug, runUrl },
   336→        },
   337→        null,
   338→        2
   339→      ) + "\n"
   340→    );
   341→
   342→    if (NO_PUSH) {
   343→      console.log("[weave-anchor-zero] recovery: --no-push — לא נדחף");
   344→      return;
   345→    }
   346→    if (!gitOrNull("git remote get-url origin")) throw new Error("אין remote origin — בלי ריפו אין עדות");
   347→    execSync(`git add -A`, { cwd: ROOT, stdio: "ignore" });
   348→    const staged = gitOrNull("git diff --cached") ?? "";
   349→    const leak = secretGate(filterKnownHashes(staged));
   350→    if (leak) {
   351→      execSync("git reset", { cwd: ROOT, stdio: "ignore" });
   352→      throw new Error(`שער הסודות חסם את הדחיפה: ${leak} ב-diff — כלום לא נדחף`);
   353→    }
   354→    const msg = `weave-anchor-zero recovery: ${RECOVER_TXID.slice(0, 18)}… · cp#${cpR.index} · att ${att.seq}`;
   355→    execSync(`git commit -m ${JSON.stringify(msg)}`, { cwd: ROOT, stdio: ["ignore", "pipe", "pipe"] });
   356→    const pushOut = gitPushRetry(ROOT, BRANCH);
   357→    console.log(`[weave-anchor-zero] recovery: ${pushOut.split("\n").slice(-1)[0] || "pushed"}`);
   358→    return;
   359→  }
   360→
   361→  // 5. ה-checkpoint האחרון — מה שעולה לשרשרת
   362→  const cp = await db.checkpoint.findFirst({ orderBy: { index: "desc" } });
   363→  if (!cp) {
   364→    throw new Error("אין checkpoint בספר — הרץ פעימה קודם (weave-beat)");
   365→  }
   366→
   367→  const line = readLine(seal.networkAddress);
   368→  const alreadyAnchored = line.last && line.last.root === cp.root;
   369→
   370→  // 6. dedupe: root זה כבר עוגן ⇒ קריאה-חזרה בלבד (השרשרת השופטת)
   371→  if (alreadyAnchored || VERIFY_ONLY) {
