    84→const f2 = (v?: number | null) => (typeof v === 'number' ? v.toFixed(2) : '—')
    85→const uptimeStr = (bootedAt?: string): string => {
    86→  if (!bootedAt) return '—'
    87→  const sec = Math.max(0, Math.floor((Date.now() - new Date(bootedAt).getTime()) / 1000))
    88→  const h = Math.floor(sec / 3600), m = Math.floor((sec % 3600) / 60)
    89→  return h > 0 ? `${h}ש ${m}דק` : `${m}דק`
    90→}
