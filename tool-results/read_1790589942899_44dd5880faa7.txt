     1→/* =====================================================================
     2→ * THE WEAVE · saos-live · the live currency protocol (saos.live.v1)
     3→ *
     4→ * What this is: SAOS lives on a public blockchain. Every operation is a
     5→ * custom_json (op id `saos.weave.live.v1`, posting authority only, so
     6→ * the chain itself forbids this key from ever moving funds). Authority
     7→ * is the inner signature of the wallet owner. The relayer is transport
     8→ * only: it cannot forge (it lacks the key) and every node re-verifies
     9→ * every signature. The same code runs in the browser, in the cloud
    10→ * heartbeat and on any machine. Same ops in, same stateRoot out.
    11→ *
    12→ * v1.1 extension (deposits, exchange, rewards, config):
    13→ *   · deposit.claim  (wallet)  : a depositor signs what they sent and
    14→ *     the reference (txid). Pending until the network treasury
    15→ *     countersigns a credit.
    16→ *   · deposit.credit (treasury): credits SAOS for a verified deposit.
    17→ *     Records currency, amount, reference and the exact SAOS amount.
    18→ *   · dex.offer      (wallet)  : sell SAOS for a listed currency at a
    19→ *     chosen price. The SAOS moves into deterministic escrow.
    20→ *   · dex.cancel     (wallet)  : returns the unlocked remainder.
    21→ *   · dex.fill       (treasury): records the external settlement
    22→ *     (payment reference) and releases the escrowed SAOS to the buyer.
    23→ *   · reward.pay     (treasury): pays SAOS rewards to a wallet, tracked
    24→ *     in the open ledger. No promises live in code, only history.
    25→ *   · cfg.addr / cfg.rate (treasury): publish deposit addresses and
    26→ *     reference rates per pair. Governance decisions, on chain.
    27→ *
    28→ * Honest trust label: relayers currently are the network witness line
    29→ * and the operator account (any Steem account may relay in future).
    30→ * A relayer cannot forge but can ignore; the public ledger is the
    31→ * judge, and settlement of external currencies is signed by the
    32→ * treasury and labeled as such. Nothing here promises profit.
    33→ * ===================================================================== */
    34→import * as GateCryptoModule from "./gate-crypto.js";
    35→const C = (typeof GateCryptoModule.default === "object" && GateCryptoModule.default) || globalThis.GateCrypto;
    36→if (!C) throw new Error("saos-live: gate-crypto missing; load gate-crypto.js with saos-live.js");
    37→
    38→function makeSaosLive(C) {
    39→  "use strict";
    40→  const PROTOCOL = "saos.live.v1";
    41→  const OP_ID = "saos.weave.live.v1";
    42→
    43→  /* Canonical JSON: recursively sorted keys, no spaces. The contract. */
    44→  function canon(v) {
    45→    if (v === null || typeof v !== "object") return JSON.stringify(v);
    46→    if (Array.isArray(v)) return "[" + v.map(canon).join(",") + "]";
    47→    const keys = Object.keys(v).sort();
    48→    return "{" + keys.map((k) => JSON.stringify(k) + ":" + canon(v[k])).join(",") + "}";
    49→  }
    50→
    51→  const utf8 = (s) => new TextEncoder().encode(s);
    52→
    53→  /* Personal address: saos1 + base58check(ripemd160(sha256(pub))).
    54→   * Self-certifying: whoever holds the private key owns the address. */
    55→  const ADDR_RE = /^saos1[1-9A-HJ-NP-Za-km-z]{33,35}$/;
    56→  async function addressFromPubSTM(stm) {
    57→    const pub33 = C.stmToPub(String(stm || ""));
    58→    const h1 = await C.sha256(pub33);
    59→    const payload = C.rmd160(h1); // 20 bytes
    60→    const chk = (await C.sha256(await C.sha256(payload))).slice(0, 4);
    61→    return "saos1" + C.b58encode(C.cat(payload, chk));
    62→  }
    63→
    64→  /* Signed message digest: sha256(UTF8(canon(envelope without sig))). */
    65→  async function digestFor(envelope) {
    66→    const { sig, ...sans } = envelope;
    67→    if (typeof sig !== "string") return C.sha256(utf8(canon(envelope)));
    68→    return C.sha256(utf8(canon(sans)));
    69→  }
    70→
    71→  async function signEnvelope(envelope, privBytes) {
    72→    const digest = await digestFor(envelope);
    73→    const sig = await C.signCompact(digest, privBytes);
    74→    return C.hex(sig);
    75→  }
    76→
    77→  function recoverPub(digest32, sigHex) {
    78→    const sig = C.unhex(String(sigHex || ""));
    79→    if (sig.length !== 65) return null;
    80→    const i = sig[0] - 31;
    81→    if (i < 0 || i > 3) return null;
    82→    const r = C.fromBE(sig.slice(1, 33));
    83→    const s = C.fromBE(sig.slice(33, 65));
    84→    const e = C.fromBE(digest32);
    85→    const pub = C._dbg.tryRecover(e, r, s, i);
    86→    return pub;
    87→  }
    88→
    89→  /* Integer amount strings (BigInt inside, no float). */
    90→  const AMOUNT_RE = /^\d{1,18}$/;
    91→  const parseAmount = (s) => (AMOUNT_RE.test(String(s)) ? BigInt(s) : null);
    92→  const fmt = (units, decimals) => {
    93→    const d = Number(decimals) || 0;
    94→    if (d === 0) return String(units);
    95→    let s = String(units).padStart(d + 1, "0");
    96→    const cut = s.length - d;
    97→    const whole = s.slice(0, cut);
    98→    let frac = s.slice(cut).replace(/0+$/, "");
    99→    return frac ? `${whole}.${frac}` : whole;
   100→  };
   101→
   102→  /* Decimal display strings for external currencies and prices.
   103→   * Not consensus money: SAOS math stays integer. */
   104→  const DEC_RE = /^(0|[1-9]\d{0,11})(\.\d{1,8})?$/;
   105→  const isDec = (s) => typeof s === "string" && DEC_RE.test(s) && !/^0+(\.0+)?$/.test(s);
   106→  const CUR_RE = /^[A-Z]{2,10}$/;
   107→  const PAIR_RE = /^[A-Z]{2,10}\/[A-Z]{2,10}$/;
   108→  const REF_RE = /^[\w :.\-\/+=]{6,90}$/;
   109→  const IDREF_RE = /^[0-9a-zA-Z._-]{6,40}$/;
   110→
   111→  /* Operation id: <milliseconds>-<random hex> */
   112→  const ID_RE = /^[0-9]{12,14}-[0-9a-f]{8,12}$/;
   113→  const GENESIS_ID_RE = /^genesis-saos-[a-z0-9]{2,12}$/;
   114→  function makeId() {
   115→    const rnd = new Uint8Array(6);
   116→    globalThis.crypto.getRandomValues(rnd);
   117→    return Date.now() + "-" + C.hex(rnd);
   118→  }
   119→
   120→  /* Genesis spec: initial mint mirrors the verified v0 supply 1:1
   121→   * (126,682 SAOS, issuer headcorner, genesis 2026-09-05). Hard cap
   122→   * 21,000,000 SAOS, fixed in the code every node runs. */
   123→  const SAOS_GENESIS = {
   124→    symbol: "SAOS",
   125→    name: "SAOS",
   126→    decimals: 3,
   127→    maxSupply: "21000000000", // 21,000,000.000 SAOS
   128→    initialMint: "126682000", // 126,682.000 SAOS, mirrors v0 exactly
   129→    note: "initial mint mirrors SAOS-NET v0 verified supply 1:1 (126,682 SAOS, issuer headcorner, genesis 2026-09-05)",
   130→  };
   131→  const TREASURY_GENESIS_ID = "genesis-saos-treasury";
   132→
   133→  /* The rules. Every op passes: strict shape, then a valid signature,
   134→   * then type rules. A rejected op never touches state (fail-closed)
   135→   * and is recorded in invalid. Metadata keys (at, txid, block) are
   136→   * tolerated and stripped: they are not signed and never enter the
   137→   * stateRoot, so every node agrees regardless of how it read the ops. */
   138→  const KINDS = {
   139→    "wallet.create": ["nick"],
   140→    "token.deploy": ["symbol", "name", "decimals", "maxSupply"],
   141→    "token.mint": ["symbol", "to", "amount"],
   142→    "token.transfer": ["symbol", "to", "amount"],
   143→    "token.burn": ["symbol", "amount"],
   144→    "deposit.claim": ["cur", "amt", "ref"],
   145→    "deposit.credit": ["claimId", "to", "saos", "cur", "amt", "ref", "note"],
   146→    "dex.offer": ["cur", "units", "price"],
   147→    "dex.cancel": ["offerId"],
   148→    "dex.fill": ["offerId", "to", "units", "paid", "ref"],
   149→    "reward.pay": ["to", "amount", "note"],
   150→    "cfg.addr": ["cur", "addr", "memo"],
   151→    "cfg.rate": ["pair", "price"],
   152→  };
   153→  const TREASURY_KINDS = new Set(["deposit.credit", "dex.fill", "reward.pay", "cfg.addr", "cfg.rate"]);
   154→  const META_KEYS = new Set(["at", "txid", "block", "trxInBlock"]);
   155→
   156→  const MAX_OPEN_OFFERS = 30;
   157→  const MAX_PENDING_CLAIMS = 20;
   158→
   159→  function openOffersOf(state, addr) {
   160→    let n = 0;
   161→    for (const id of Object.keys(state.offers)) if (state.offers[id].by === addr && state.offers[id].status === "open") n++;
   162→    return n;
   163→  }
   164→  function pendingClaimsOf(state, addr) {
   165→    let n = 0;
   166→    for (const id of Object.keys(state.claims)) if (state.claims[id].by === addr && state.claims[id].status === "pending") n++;
   167→    return n;
   168→  }
   169→
   170→  async function validateEnvelope(rawEnv, state, seenIds, seenPubs) {
   171→    if (!rawEnv || typeof rawEnv !== "object") return { err: "bad-envelope" };
   172→    // strip metadata (not signed, not consensus) before any check
   173→    const env = {};
   174→    for (const key of Object.keys(rawEnv)) if (!META_KEYS.has(key)) env[key] = rawEnv[key];
   175→
   176→    if (env.v !== 1) return { err: "version" };
   177→    const k = String(env.k || "");
   178→    if (!KINDS[k]) return { err: "unknown-kind" };
   179→    const id = String(env.id || "");
   180→    if (!(ID_RE.test(id) || GENESIS_ID_RE.test(id))) return { err: "bad-id" };
   181→    if (seenIds.has(id)) return { err: "dup-id" };
   182→    if (typeof env.pub !== "string" || !/^STM[1-9A-HJ-NP-Za-km-z]{50}$/.test(env.pub)) return { err: "bad-pub" };
   183→    if (typeof env.sig !== "string" || !/^[0-9a-f]{130}$/.test(env.sig)) return { err: "bad-sig" };
   184→    if (typeof env.by !== "string" || !ADDR_RE.test(env.by)) return { err: "bad-addr" };
   185→
   186→    let pub33;
   187→    try {
   188→      pub33 = C.stmToPub(env.pub);
   189→    } catch {
   190→      return { err: "bad-pub" };
   191→    }
   192→    const digest = await digestFor(env);
   193→    const rec = recoverPub(digest, env.sig);
   194→    if (!rec || C.hex(rec) !== C.hex(pub33)) return { err: "sig" };
   195→    const derived = await addressFromPubSTM(env.pub);
   196→    if (derived !== env.by) return { err: "addr-mismatch" };
   197→
   198→    // strict fields: unknown field means rejection, the canon is the contract
   199→    for (const key of Object.keys(env)) {
   200→      const allowed = new Set(["v", "id", "k", "by", "pub", "sig", ...KINDS[k]]);
   201→      if (!allowed.has(key)) return { err: "unexpected-field:" + key };
   202→    }
   203→    const optStr = (v, max) => v === undefined || (typeof v === "string" && v.length >= 1 && v.length <= max);
   204→
   205→    if (k === "wallet.create") {
   206→      if (state.wallets[env.by]) return { err: "wallet-exists" };
   207→      if (seenPubs.has(env.pub)) return { err: "pub-taken" };
   208→      if (env.nick !== undefined) {
   209→        if (typeof env.nick !== "string" || env.nick.length < 1 || env.nick.length > 24) return { err: "bad-nick" };
   210→        if (!/^[\u0590-\u05FFa-zA-Z0-9 ._-]+$/.test(env.nick)) return { err: "bad-nick" };
   211→      }
   212→      return { ok: true };
   213→    }
   214→
   215→    // every other kind requires a registered wallet
   216→    const sender = state.wallets[env.by];
   217→    if (!sender) return { err: "no-wallet" };
   218→
   219→    // treasury-only kinds: authority is the genesis treasury address
   220→    if (TREASURY_KINDS.has(k)) {
   221→      if (!state.treasury) return { err: "no-treasury" };
   222→      if (env.by !== state.treasury) return { err: "not-treasury" };
   223→    }
   224→
   225→    if (k === "token.deploy") {
   226→      const symbol = String(env.symbol || "");
   227→      if (!/^[A-Z][A-Z0-9]{2,7}$/.test(symbol)) return { err: "bad-symbol" };
   228→      if (state.tokens[symbol]) return { err: "symbol-taken" };
   229→      if (typeof env.name !== "string" || env.name.length < 1 || env.name.length > 48) return { err: "bad-name" };
   230→      if (!Number.isInteger(env.decimals) || env.decimals < 0 || env.decimals > 9) return { err: "bad-decimals" };
   231→      const max = parseAmount(env.maxSupply);
   232→      if (max === null || max <= 0n || max > 10n ** 15n) return { err: "bad-max" };
   233→      return { ok: true };
   234→    }
   235→
   236→    if (k === "deposit.claim") {
   237→      if (!CUR_RE.test(String(env.cur || ""))) return { err: "bad-cur" };
   238→      if (!isDec(env.amt)) return { err: "bad-amt" };
   239→      if (!REF_RE.test(String(env.ref || ""))) return { err: "bad-ref" };
   240→      if (pendingClaimsOf(state, env.by) >= MAX_PENDING_CLAIMS) return { err: "too-many-claims" };
   241→      return { ok: true };
   242→    }
   243→
   244→    if (k === "deposit.credit") {
   245→      if (env.claimId !== undefined && !IDREF_RE.test(String(env.claimId))) return { err: "bad-claimId" };
   246→      if (typeof env.to !== "string" || !ADDR_RE.test(env.to) || !state.wallets[env.to]) return { err: "no-recipient" };
   247→      const a = parseAmount(env.saos);
   248→      if (a === null || a <= 0n) return { err: "bad-saos" };
   249→      if (!CUR_RE.test(String(env.cur || ""))) return { err: "bad-cur" };
   250→      if (!isDec(env.amt)) return { err: "bad-amt" };
   251→      if (!optStr(env.ref, 90) || (env.ref !== undefined && !REF_RE.test(env.ref))) return { err: "bad-ref" };
   252→      if (!optStr(env.note, 120)) return { err: "bad-note" };
   253→      if (env.claimId !== undefined) {
   254→        const cl = state.claims[String(env.claimId)];
   255→        if (!cl) return { err: "no-claim" };
   256→        if (cl.status !== "pending") return { err: "claim-not-pending" };
   257→      }
   258→      if ((state.bal[env.by]?.SAOS || 0n) < a) return { err: "insufficient" };
   259→      return { ok: true };
   260→    }
   261→
   262→    if (k === "dex.offer") {
   263→      if (!state.tokens.SAOS) return { err: "no-saos" };
   264→      if (!CUR_RE.test(String(env.cur || ""))) return { err: "bad-cur" };
   265→      const u = parseAmount(env.units);
   266→      if (u === null || u <= 0n) return { err: "bad-units" };
   267→      if (!isDec(env.price)) return { err: "bad-price" };
   268→      if ((state.bal[env.by]?.SAOS || 0n) < u) return { err: "insufficient" };
   269→      if (openOffersOf(state, env.by) >= MAX_OPEN_OFFERS) return { err: "too-many-offers" };
   270→      return { ok: true };
   271→    }
   272→
   273→    if (k === "dex.cancel") {
   274→      if (!IDREF_RE.test(String(env.offerId || ""))) return { err: "bad-offerId" };
   275→      const off = state.offers[String(env.offerId)];
   276→      if (!off) return { err: "no-offer" };
   277→      if (off.by !== env.by) return { err: "not-owner" };
   278→      if (off.status !== "open") return { err: "not-open" };
   279→      return { ok: true };
   280→    }
   281→
   282→    if (k === "dex.fill") {
   283→      if (!IDREF_RE.test(String(env.offerId || ""))) return { err: "bad-offerId" };
   284→      const off = state.offers[String(env.offerId)];
   285→      if (!off) return { err: "no-offer" };
   286→      if (off.status !== "open") return { err: "not-open" };
   287→      if (typeof env.to !== "string" || !ADDR_RE.test(env.to) || !state.wallets[env.to]) return { err: "no-recipient" };
   288→      const u = parseAmount(env.units);
   289→      if (u === null || u <= 0n) return { err: "bad-units" };
   290→      if (u > off.units - off.filled) return { err: "over-fill" };
   291→      if (env.paid !== undefined && !isDec(env.paid)) return { err: "bad-paid" };
   292→      if (!optStr(env.ref, 90) || (env.ref !== undefined && !REF_RE.test(env.ref))) return { err: "bad-ref" };
   293→      return { ok: true };
   294→    }
   295→
   296→    if (k === "reward.pay") {
   297→      if (typeof env.to !== "string" || !ADDR_RE.test(env.to) || !state.wallets[env.to]) return { err: "no-recipient" };
   298→      const a = parseAmount(env.amount);
   299→      if (a === null || a <= 0n) return { err: "bad-amount" };
   300→      if (!optStr(env.note, 120)) return { err: "bad-note" };
   301→      if ((state.bal[env.by]?.SAOS || 0n) < a) return { err: "insufficient" };
   302→      return { ok: true };
   303→    }
   304→
   305→    if (k === "cfg.addr") {
   306→      if (!CUR_RE.test(String(env.cur || ""))) return { err: "bad-cur" };
   307→      if (typeof env.addr !== "string" || env.addr.length < 8 || env.addr.length > 120) return { err: "bad-addr-str" };
   308→      if (!optStr(env.memo, 80)) return { err: "bad-memo" };
   309→      return { ok: true };
   310→    }
   311→
   312→    if (k === "cfg.rate") {
   313→      if (!PAIR_RE.test(String(env.pair || ""))) return { err: "bad-pair" };
   314→      if (!isDec(env.price)) return { err: "bad-price" };
   315→      return { ok: true };
   316→    }
   317→
   318→    const symbol = String(env.symbol || "");
   319→    const token = state.tokens[symbol];
   320→    if (!token) return { err: "no-token" };
   321→
   322→    if (k === "token.mint") {
   323→      if (env.by !== token.issuer) return { err: "not-issuer" };
   324→      if (typeof env.to !== "string" || !ADDR_RE.test(env.to) || !state.wallets[env.to]) return { err: "no-recipient" };
   325→      const a = parseAmount(env.amount);
   326→      if (a === null || a <= 0n) return { err: "bad-amount" };
   327→      if ((state.supply[symbol] || 0n) + a > BigInt(token.maxSupply)) return { err: "over-max" };
   328→      return { ok: true };
   329→    }
   330→
   331→    if (k === "token.transfer") {
   332→      if (typeof env.to !== "string" || !ADDR_RE.test(env.to) || !state.wallets[env.to]) return { err: "no-recipient" };
   333→      if (env.to === env.by) return { err: "self" };
   334→      const a = parseAmount(env.amount);
   335→      if (a === null || a <= 0n) return { err: "bad-amount" };
   336→      if ((state.bal[env.by]?.[symbol] || 0n) < a) return { err: "insufficient" };
   337→      return { ok: true };
   338→    }
   339→
   340→    if (k === "token.burn") {
   341→      const a = parseAmount(env.amount);
   342→      if (a === null || a <= 0n) return { err: "bad-amount" };
   343→      if ((state.bal[env.by]?.[symbol] || 0n) < a) return { err: "insufficient" };
   344→      return { ok: true };
   345→    }
   346→
   347→    return { err: "unreachable" };
   348→  }
   349→
   350→  /* The fold: a sequence of ops becomes a state. Fully deterministic:
   351→   * same ops in the same order give the same stateRoot. Invalid ops
   352→   * are recorded and skipped; they never touch state. */
   353→  async function fold(envelopes) {
   354→    const state = {
   355→      wallets: {}, // addr -> {pub, nick, at, txid}
   356→      tokens: {}, // SYMBOL -> {name, decimals, maxSupply, issuer, at, txid}
   357→      bal: {}, // addr -> {SYMBOL: BigInt}
   358→      supply: {}, // SYMBOL -> BigInt
   359→      claims: {}, // claimId -> {by, cur, amt, ref, status, at, txid}
   360→      offers: {}, // offerId -> {by, cur, units, price, filled, status, fills[], at, txid}
   361→      cfg: { addrs: {}, rates: {} }, // treasury published config
   362→      rewardsPaid: 0n, // total SAOS units paid as rewards
   363→      treasury: null, // genesis treasury address
   364→    };
   365→    const seenIds = new Set();
   366→    const seenPubs = new Set();
   367→    const invalid = [];
   368→    const order = [];
   369→    let valid = 0;
   370→
   371→    const meta = (env) => ({
   372→      at: typeof env?.at === "string" ? env.at : null,
   373→      txid: typeof env?.txid === "string" ? env.txid : null,
   374→    });
   375→
   376→    const list = Array.isArray(envelopes) ? envelopes.slice() : [];
   377→    for (const rawEnv of list) {
   378→      const m = meta(rawEnv);
   379→      const res = await validateEnvelope(rawEnv, state, seenIds, seenPubs);
   380→      if (res.err) {
   381→        invalid.push({ id: String(rawEnv?.id ?? "?"), k: String(rawEnv?.k ?? "?"), err: res.err });
   382→        continue;
   383→      }
   384→      // clean view for the executor (metadata stripped)
   385→      const env = {};
   386→      for (const key of Object.keys(rawEnv)) if (!META_KEYS.has(key)) env[key] = rawEnv[key];
   387→
   388→      seenIds.add(String(env.id));
   389→      seenPubs.add(String(env.pub));
   390→      const k = String(env.k);
   391→      if (k === "wallet.create") {
   392→        state.wallets[env.by] = { pub: env.pub, nick: typeof env.nick === "string" ? env.nick : null, ...m };
   393→        if (env.id === TREASURY_GENESIS_ID) state.treasury = env.by;
   394→      } else if (k === "token.deploy") {
   395→        state.tokens[env.symbol] = { name: env.name, decimals: env.decimals, maxSupply: String(env.maxSupply), issuer: env.by, ...m };
   396→      } else if (k === "token.mint") {
   397→        state.bal[env.to] = state.bal[env.to] || {};
   398→        state.bal[env.to][env.symbol] = (state.bal[env.to][env.symbol] || 0n) + BigInt(env.amount);
   399→        state.supply[env.symbol] = (state.supply[env.symbol] || 0n) + BigInt(env.amount);
   400→      } else if (k === "token.transfer") {
   401→        state.bal[env.by][env.symbol] -= BigInt(env.amount);
   402→        state.bal[env.to] = state.bal[env.to] || {};
   403→        state.bal[env.to][env.symbol] = (state.bal[env.to][env.symbol] || 0n) + BigInt(env.amount);
   404→      } else if (k === "token.burn") {
   405→        state.bal[env.by][env.symbol] -= BigInt(env.amount);
   406→        state.supply[env.symbol] -= BigInt(env.amount);
   407→        if (state.bal[env.by][env.symbol] === 0n) delete state.bal[env.by][env.symbol];
   408→      } else if (k === "deposit.claim") {
   409→        state.claims[env.id] = { by: env.by, cur: env.cur, amt: env.amt, ref: env.ref, status: "pending", ...m };
   410→      } else if (k === "deposit.credit") {
   411→        state.bal[env.by].SAOS -= BigInt(env.saos);
   412→        state.bal[env.to] = state.bal[env.to] || {};
   413→        state.bal[env.to].SAOS = (state.bal[env.to].SAOS || 0n) + BigInt(env.saos);
   414→        if (env.claimId !== undefined) {
   415→          const cl = state.claims[String(env.claimId)];
   416→          if (cl) cl.status = "credited";
   417→        }
   418→      } else if (k === "dex.offer") {
   419→        state.bal[env.by].SAOS -= BigInt(env.units); // into deterministic escrow
   420→        state.offers[env.id] = { by: env.by, cur: env.cur, units: BigInt(env.units), price: env.price, filled: 0n, status: "open", fills: [], ...m };
   421→      } else if (k === "dex.cancel") {
   422→        const off = state.offers[String(env.offerId)];
   423→        const back = off.units - off.filled;
   424→        state.bal[off.by] = state.bal[off.by] || {};
   425→        state.bal[off.by].SAOS = (state.bal[off.by].SAOS || 0n) + back;
   426→        off.status = "cancelled";
   427→      } else if (k === "dex.fill") {
   428→        const off = state.offers[String(env.offerId)];
   429→        const u = BigInt(env.units);
   430→        off.filled += u;
   431→        off.fills.push({ to: env.to, units: String(env.units), paid: env.paid !== undefined ? env.paid : null, ref: env.ref !== undefined ? env.ref : null, at: m.at });
   432→        state.bal[env.to] = state.bal[env.to] || {};
   433→        state.bal[env.to].SAOS = (state.bal[env.to].SAOS || 0n) + u;
   434→        if (off.filled === off.units) off.status = "closed";
   435→      } else if (k === "reward.pay") {
   436→        state.bal[env.by].SAOS -= BigInt(env.amount);
   437→        state.bal[env.to] = state.bal[env.to] || {};
   438→        state.bal[env.to].SAOS = (state.bal[env.to].SAOS || 0n) + BigInt(env.amount);
   439→        state.rewardsPaid += BigInt(env.amount);
   440→      } else if (k === "cfg.addr") {
   441→        state.cfg.addrs[env.cur] = { addr: env.addr, memo: env.memo !== undefined ? env.memo : null, ...m };
   442→      } else if (k === "cfg.rate") {
   443→        state.cfg.rates[env.pair] = { price: env.price, ...m };
   444→      }
   445→      valid++;
   446→      order.push({ id: env.id, k, by: env.by, ...m });
   447→    }
   448→
   449→    /* stateRoot projection: consensus fields only. Metadata (at, txid)
   450→     * never enters, so cloud, browser and any node agree byte for byte. */
   451→    const balCanon = {};
   452→    for (const addr of Object.keys(state.bal).sort()) {
   453→      const entries = Object.entries(state.bal[addr]).filter(([, v]) => v > 0n);
   454→      if (entries.length) balCanon[addr] = Object.fromEntries(entries.map(([s, v]) => [s, String(v)]));
   455→    }
   456→    const supplyCanon = Object.fromEntries(Object.entries(state.supply).map(([s, v]) => [s, String(v)]));
   457→    const walletsCanon = Object.fromEntries(
   458→      Object.keys(state.wallets).sort().map((a) => [a, { pub: state.wallets[a].pub, nick: state.wallets[a].nick ?? null }]),
   459→    );
   460→    const tokensCanon = Object.fromEntries(
   461→      Object.keys(state.tokens).sort().map((s) => {
   462→        const t = state.tokens[s];
   463→        return [s, { name: t.name, decimals: t.decimals, maxSupply: t.maxSupply, issuer: t.issuer }];
   464→      }),
   465→    );
   466→    const claimsCanon = Object.fromEntries(
   467→      Object.keys(state.claims).sort().map((id) => {
   468→        const c = state.claims[id];
   469→        return [id, { by: c.by, cur: c.cur, amt: c.amt, ref: c.ref, status: c.status }];
   470→      }),
   471→    );
   472→    const offersCanon = Object.fromEntries(
   473→      Object.keys(state.offers).sort().map((id) => {
   474→        const o = state.offers[id];
   475→        // fills enter the root without their `at` metadata: consensus fields only
   476→        const fillsCanon = o.fills.map((f) => ({ to: f.to, units: f.units, paid: f.paid ?? null, ref: f.ref ?? null }));
   477→        return [id, { by: o.by, cur: o.cur, units: String(o.units), price: o.price, filled: String(o.filled), status: o.status, fills: fillsCanon }];
   478→      }),
   479→    );
   480→    const cfgCanon = {
   481→      addrs: Object.fromEntries(Object.keys(state.cfg.addrs).sort().map((cur) => {
   482→        const a = state.cfg.addrs[cur];
   483→        return [cur, { addr: a.addr, memo: a.memo ?? null }];
   484→      })),
   485→      rates: Object.fromEntries(Object.keys(state.cfg.rates).sort().map((pair) => {
   486→        const r = state.cfg.rates[pair];
   487→        return [pair, { price: r.price }];
   488→      })),
   489→    };
   490→    const rootInput = canon({
   491→      wallets: walletsCanon,
   492→      tokens: tokensCanon,
   493→      balances: balCanon,
   494→      supply: supplyCanon,
   495→      claims: claimsCanon,
   496→      offers: offersCanon,
   497→      cfg: cfgCanon,
   498→      rewardsPaid: String(state.rewardsPaid),
   499→      ops: valid,
   500→    });
   501→    const stateRoot = C.hex(await C.sha256(utf8(rootInput)));
   502→
   503→    return {
   504→      ok: true,
   505→      protocol: PROTOCOL,
   506→      valid,
   507→      invalid,
   508→      order,
   509→      state,
   510→      stateRoot,
   511→      balCanon,
   512→      supplyCanon,
   513→      walletsCanon,
   514→      tokensCanon,
   515→      claimsCanon,
   516→      offersCanon,
   517→      cfgCanon,
   518→    };
   519→  }

   521→  /* Build a signed op (the wallet in the browser). */
   522→  async function buildOp(privBytes, pubSTM, fields) {
   523→    const by = await addressFromPubSTM(pubSTM);
   524→    const env = { v: 1, id: makeId(), by, pub: pubSTM, ...fields };
   525→    env.sig = await signEnvelope(env, privBytes);
   526→    return env;
   527→  }

   529→  /* Genesis (cloud only): register treasury, deploy SAOS, first mint. */
   530→  async function buildGenesisOps(treasuryPrivBytes, treasuryPubSTM) {
   531→    const treasury = await addressFromPubSTM(treasuryPubSTM);
   532→    const wallet = {
   533→      v: 1,
   534→      id: TREASURY_GENESIS_ID,
   535→      k: "wallet.create",
   536→      nick: "network-treasury",
   537→      by: treasury,
   538→      pub: treasuryPubSTM,
   539→    };
   540→    wallet.sig = await signEnvelope(wallet, treasuryPrivBytes);
   541→    const deploy = {
   542→      v: 1,
   543→      id: "genesis-saos-deploy",
   544→      k: "token.deploy",
   545→      symbol: SAOS_GENESIS.symbol,
   546→      name: SAOS_GENESIS.name,
   547→      decimals: SAOS_GENESIS.decimals,
   548→      maxSupply: SAOS_GENESIS.maxSupply,
   549→      by: treasury,
   550→      pub: treasuryPubSTM,
   551→    };
   552→    deploy.sig = await signEnvelope(deploy, treasuryPrivBytes);
   553→    const mint = {
   554→      v: 1,
   555→      id: "genesis-saos-mint",
   556→      k: "token.mint",
   557→      symbol: SAOS_GENESIS.symbol,
   558→      to: treasury,
   559→      amount: SAOS_GENESIS.initialMint,
   560→      by: treasury,
   561→      pub: treasuryPubSTM,
   562→    };
   563→    mint.sig = await signEnvelope(mint, treasuryPrivBytes);
   564→    return { treasury, ops: [wallet, deploy, mint] };
   565→  }

   573→  async function newWallet() {
   574→    const priv = randomPriv();
   575→    const pub33 = C.privToPubBytes(priv);
   576→    const pubSTM = await C.pubToSTM(pub33);
   577→    const addr = await addressFromPubSTM(pubSTM);
   578→    return { priv, pubSTM, addr };

   581→  /* Self-test: fixed deterministic vectors. Test keys derive from
   582→   * sha256 of fixed strings, so no hex constants live in the code. */
   583→  async function selfTest() {
   584→    const tests = [];
   585→    const t = (name, pass, detail) => tests.push({ name, pass, detail: detail || null });
   586→
   587→    t("canon: recursive key sort", canon({ b: 1, a: { z: 2, y: 3 } }) === '{"a":{"y":3,"z":2},"b":1}');
   588→
   589→    const priv1 = await C.sha256(utf8("saos-live-selftest-key-one"));
   590→    const priv2 = await C.sha256(utf8("saos-live-selftest-key-two"));
   591→    const pub1 = await C.pubToSTM(C.privToPubBytes(priv1));
   592→    const pub2 = await C.pubToSTM(C.privToPubBytes(priv2));
   593→    const addr1a = await addressFromPubSTM(pub1);
   594→    const addr1b = await addressFromPubSTM(pub1);
   595→    const addr2 = await addressFromPubSTM(pub2);
   596→    t("address: deterministic", addr1a === addr1b);
   597→    t("address: saos1 format", ADDR_RE.test(addr1a), addr1a);
   598→    t("address: different keys give different addresses", addr1a !== addr2);
   599→
   600→    const w1 = { priv: priv1, pub: pub1 };
   601→    const w2 = { priv: priv2, pub: pub2 };
   602→    const ops = [];
   603→    const mk = async (w, fields, id) => {
   604→      const by = await addressFromPubSTM(w.pub);
   605→      const env = { v: 1, id: id || makeId(), by, pub: w.pub, ...fields };
   606→      env.sig = await signEnvelope(env, w.priv);
   607→      ops.push(env);
   608→      return env;
   609→    };
   610→    await mk(w1, { k: "wallet.create", nick: "selftest-a" }, "1731000000000-aa000001");
   611→    await mk(w2, { k: "wallet.create", nick: "selftest-b" }, "1731000000001-aa000002");
   612→    await mk(w1, { k: "token.deploy", symbol: "TEST", name: "Self Test", decimals: 3, maxSupply: "1000000" }, "1731000000002-aa000003");
   613→    await mk(w1, { k: "token.mint", symbol: "TEST", to: addr1a, amount: "1000" }, "1731000000003-aa000004");
   614→    await mk(w1, { k: "token.mint", symbol: "TEST", to: addr2, amount: "500" }, "1731000000004-aa000005");
   615→    await mk(w1, { k: "token.transfer", symbol: "TEST", to: addr2, amount: "200" }, "1731000000005-aa000006");
   616→    await mk(w2, { k: "token.burn", symbol: "TEST", amount: "100" }, "1731000000006-aa000007");
   617→
   618→    const res = await fold(ops);
   619→    t("fold: 7/7 valid", res.valid === 7 && res.invalid.length === 0, JSON.stringify(res.invalid));
   620→    t("balance: sender 800", res.state.bal[addr1a]?.TEST === 800n);
   621→    t("balance: receiver 600", res.state.bal[addr2]?.TEST === 600n);
   622→    t("supply: 1400 (1500 minus 100 burned)", res.state.supply.TEST === 1400n);
   623→
   624→    const forged = { v: 1, id: "1731000000007-bb000001", k: "token.transfer", symbol: "TEST", to: addr2, amount: "50", by: addr1a, pub: pub1 };
   625→    forged.sig = await signEnvelope(forged, w2.priv);
   626→    const r2 = await fold([...ops, forged]);
   627→    t("forgery: someone else's signature is rejected", r2.valid === 7 && r2.invalid[0]?.err === "sig");
   628→
   629→    const over = await mk(w1, { k: "token.mint", symbol: "TEST", to: addr1a, amount: "999999000" }, "1731000000008-bb000002");
   630→    const r3 = await fold([...ops, over]);
   631→    t("mint above the cap is rejected", r3.invalid.some((x) => x.err === "over-max") && r3.state.supply.TEST === 1400n);
   632→
   633→    const dup = { ...ops[0] };
   634→    const r4 = await fold([...ops, dup]);
   635→    t("duplicate id is rejected", r4.valid === 7 && r4.invalid.some((x) => x.err === "dup-id"));
   636→
   637→    const ghost = await mk(w1, { k: "token.transfer", symbol: "TEST", to: "saos1" + "2".repeat(34), amount: "1" }, "1731000000009-bb000003");
   638→    const r5 = await fold([...ops, ghost]);
   639→    t("transfer to an unregistered address is rejected", r5.invalid.some((x) => x.err === "no-recipient"));
   640→
   641→    const ra = await fold(ops);
   642→    const rb = await fold(ops.slice());
   643→    t("stateRoot: deterministic across runs", ra.stateRoot === rb.stateRoot, ra.stateRoot);
   644→    t("stateRoot: 64 hex", /^[0-9a-f]{64}$/.test(ra.stateRoot));
   645→
   646→    const gen = await buildGenesisOps(priv1, pub1);
   647→    t("genesis: three signed ops", gen.ops.length === 3 && gen.ops.every((o) => /^[0-9a-f]{130}$/.test(o.sig)));
   648→    const rg = await fold(gen.ops);
   649→    t("genesis: SAOS deployed, supply 126,682", rg.valid === 3 && rg.state.supply.SAOS === 126682000n && rg.state.bal[gen.treasury].SAOS === 126682000n);
   650→    t("genesis: treasury recorded", rg.state.treasury === gen.treasury);
   651→
   652→    // metadata tolerance: chain-reading flows attach at/txid. Validation
   653→    // ignores them and the stateRoot is identical to a clean fold.
   654→    const withMeta = ops.map((o, i) => ({ ...o, at: "2026-09-15T00:00:0" + (i % 10) + "Z", txid: "abc123", block: 1000 + i }));
   655→    const rm = await fold(withMeta);
   656→    t("metadata: at/txid tolerated, same valid count", rm.valid === ra.valid, rm.valid + "/" + ra.valid);
   657→    t("metadata: same stateRoot as clean fold", rm.stateRoot === ra.stateRoot);
   658→
   659→    // deposits, exchange, rewards: treasury = w1 (genesis signer)
   660→    const sc = []; // scenario ops
   661→    const mk2 = async (w, fields, id) => {
   662→      const by = await addressFromPubSTM(w.pub);
   663→      const env = { v: 1, id, by, pub: w.pub, ...fields };
   664→      env.sig = await signEnvelope(env, w.priv);
   665→      sc.push(env);
   666→      return env;
   667→    };
   668→    await mk2(w2, { k: "wallet.create", nick: "friend" }, "1731000000009-cc000000");
   669→    const claim = await mk2(w2, { k: "deposit.claim", cur: "USD", amt: "25", ref: "tx-test-123456" }, "1731000000010-cc000001");
   670→    await mk2(w1, { k: "deposit.credit", claimId: claim.id, to: addr2, saos: "25000", cur: "USD", amt: "25", ref: "tx-test-123456" }, "1731000000011-cc000002");
   671→    await mk2(w1, { k: "cfg.addr", cur: "USD", addr: "example-payment-address-0001" }, "1731000000012-cc000003");
   672→    await mk2(w1, { k: "cfg.rate", pair: "SAOS/USD", price: "0.001" }, "1731000000013-cc000004");
   673→    const off = await mk2(w2, { k: "dex.offer", cur: "USD", units: "10000", price: "0.0012" }, "1731000000014-cc000005");
   674→    await mk2(w1, { k: "dex.fill", offerId: off.id, to: addr2, units: "4000", paid: "4.8", ref: "pay-test-654321" }, "1731000000015-cc000006");
   675→    await mk2(w2, { k: "dex.cancel", offerId: off.id }, "1731000000016-cc000007");
   676→    await mk2(w1, { k: "reward.pay", to: addr2, amount: "500", note: "first reward" }, "1731000000017-cc000008");
   677→
   678→    const full = [...gen.ops, ...sc];
   679→    const rs = await fold(full);
   680→    t("scenario: 12/12 valid (genesis + wallet + deposit + dex + reward)", rs.valid === 12 && rs.invalid.length === 0, JSON.stringify(rs.invalid));
   681→    t("claim: recorded and credited", rs.state.claims[claim.id]?.status === "credited" && rs.state.claims[claim.id]?.by === addr2);
   682→    t("credit: SAOS balance of depositor 25000 + 500 reward", rs.state.bal[addr2]?.SAOS === 25500n);
   683→    t("dex: escrow math 10000 locked, 4000 filled, 6000 returned", rs.state.offers[off.id]?.status === "cancelled" && rs.state.offers[off.id]?.filled === 4000n);
   684→    t("dex: fill recorded with payment reference", rs.state.offers[off.id]?.fills?.[0]?.ref === "pay-test-654321");
   685→    t("rewards: total paid tracked", rs.state.rewardsPaid === 500n);
   686→    t("cfg: deposit address and rate published", rs.state.cfg.addrs.USD?.addr === "example-payment-address-0001" && rs.state.cfg.rates["SAOS/USD"]?.price === "0.001");
   687→
   688→    // treasury authority: w2 tries to credit a deposit and fill an offer
   689→    const forgedCredit = await mk2(w2, { k: "deposit.credit", to: addr2, saos: "100", cur: "USD", amt: "1", ref: "forged-ref-999" }, "1731000000018-dd000001");
   690→    const forgedFill = await mk2(w2, { k: "dex.fill", offerId: off.id, to: addr2, units: "100" }, "1731000000019-dd000002");
   691→    const rf = await fold([...full, forgedCredit, forgedFill]);
   692→    t("authority: non-treasury credit rejected", rf.invalid.some((x) => x.err === "not-treasury"));
   693→    t("authority: non-treasury fill rejected", rf.invalid.filter((x) => x.err === "not-treasury").length === 2);
   694→
   695→    // offer without balance is rejected
   696→    const bigOffer = await mk2(w2, { k: "dex.offer", cur: "USD", units: "99999999", price: "1" }, "1731000000020-dd000003");
   697→    const ro = await fold([...full, bigOffer]);
   698→    t("dex: offer beyond balance rejected", ro.invalid.some((x) => x.err === "insufficient"));
   699→
   700→    // over-fill is rejected
   701→    const overFill = await mk2(w1, { k: "dex.fill", offerId: off.id, to: addr2, units: "999999" }, "1731000000021-dd000004");
   702→    const rOF = await fold([...full, overFill]);
   703→    t("dex: fill above remaining rejected (not-open after cancel)", rOF.invalid.some((x) => x.err === "not-open" || x.err === "over-fill"));
   704→
   705→    const passed = tests.filter((x) => x.pass).length;
   706→    return { passed, total: tests.length, tests, protocol: PROTOCOL };
   707→  }
