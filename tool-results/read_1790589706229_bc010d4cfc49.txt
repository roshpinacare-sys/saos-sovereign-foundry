   430→async function seal(rt: ChainRT, blockRoot: string): Promise<void> {
   431→  const st = rt.st;
   432→  if (rt.blockHashes.length === 0) return;
   433→  // FIX(Roast-65-a · HIGH-2): ביקורת-השימור רצה-עתה בתוך-החותם, לפני-כל-פרסום.
   434→  // הישן: invariant() נקרא רק בנתיבי-ספר/בריכה (טענות-צרות) ו-ledgerAudit רץ רק ב-GET

   436→  // חותמות "תקינות" על מצב-מנופח ושרשרת-העדויות כולה מאמתת את השקר. עכשיו: !ok =
   437→  // InvariantViolation → HALT עם-פסק-דין-פורנזי (אין-חותמות-שלמות על-לדגר-שבור).
   438→  // ייבוא-דינמי — settle.ts מייבא-מ-chain.ts (getWallets) ואנחנו לא יוצרים-מעגל-סטטי.
   439→  const { ledgerAudit, genesisTotals } = await import("./settle");
   440→  const audit = ledgerAudit(st, genesisTotals());
   441→  if (!audit.ok) {
   442→    const forensic = [...audit.errors, ...audit.negatives].slice(0, 3).join(" | ") || "unknown";
   443→    throw new InvariantViolation(`ledger-audit-failed@seal h${st.height}: ${forensic}`);
   444→  }
   445→  const root = merkleRoot(rt.blockHashes);
   446→  const info: SealInfo = {
   447→    fromHeight: st.height - rt.blockHashes.length + 1,
   448→    toHeight: st.height,
   449→    root, stateRoot: stateRoot(st),
   450→    leaves: rt.blockHashes.length,
   451→    at: Date.now(),
   452→    digest: "",
   453→    // FIX(Roast-65-a · MED-7): החלון-החתום עצמו — העדות למה שהשורש מחייב (היה-רק count)
   454→    blockHashes: [...rt.blockHashes],
   455→  };
   456→  info.digest = sha256(canon(info));
   457→  /* ── Formal Assertions (Fail-Fast) — חוקי-הברזל לפני-פרסום-החותם ──
   458→   * החותם הוא העדות-הקריפטוגרפית היוצאת-מהמערכת; אי-עקביות בין החלון,
   459→   * שורש-המרקל וה-stateRoot = עדות-שקרית בפוטנציה. הבדיקות מתבצעות לפני
   460→   * כל-פרסום (מדדים/DB/Relay) — ומפרסמות fail-closed: אם אחת-מהן נופלת,
   461→   * שום-דבר לא יוצא, חלון-הבלוקים נשמר, והחותם-הבא ינסה-שוב (ריפוי-עצמי). */
   462→  // (1) החלון מכסה בדיוק את טווח-הגבהים הרציף שהחותם מצהיר עליו
   463→  invariant(info.toHeight - info.fromHeight + 1 === rt.blockHashes.length, "seal-window-contiguity");
   464→  // (2) שורש-המרקל אכן נגזר מhashי-הבלוקים שבחלון (עדות-מחייבת את הבלוקים עצמם)
   465→  invariant(merkleRoot(rt.blockHashes) === root, "seal-merkle-root");

   467→  // היה-פה חישוב-O(state)-מלא שני (טאוטולוגיה — אותה-פונקציה טהורה על אותו-מצב)

   504→/** ── SETTLE-14 PRICE-TRUTH: עדות-מחיר-אמת ──
   505→ * כל-אירוע-מסחר אמת (fill על-הספר, swap בבריכה) נרשם: מה-נמכר, באיזה-מחיר,
   506→ * כמה-ציטוט-זרם (muQ = amountMu×px/1000). הטוויאפ-השקול-נוטיונל הוא המדד
   507→ * היחיד ל"מחיר-אמת" — הספר והבריכה עלולים-לצוף (וצפו — ממצא-חי +15%/+25%),
   508→ * אבל עסקאות-שקרו-בפועל הן-העדות. חוץ-מצב (לא-בשורש) — טבעת 600. */
   509→function feedPriceOracle(rt: ChainRT, events: TradeEvent[] | undefined, h: number): void {
   510→  if (!events || events.length === 0) return;
   511→  for (const ev of events) {
   512→    if (ev.kind !== "fill" && ev.kind !== "swap") continue;
   513→    if (!(ev.price > 0) || !(ev.amountMu > 0)) continue;
   514→    rt.priceSamples.push({ h, base: ev.base, quote: ev.quote, px: ev.price, muQ: ev.quoteMu ?? Math.floor((ev.amountMu * ev.price) / 1000) }); // FIX(R15-a): quoteMu — משקל-ציטוט-אמת (קניות-בריכה היו ×price/1000-מנופפות)
   515→  }
   516→  if (rt.priceSamples.length > 600) rt.priceSamples.splice(0, rt.priceSamples.length - 600);
   517→}

   519→export interface PriceTruth { base: string; quote: string; twap: number; samples: number; devBpsVsRef: number; windowHeights: number }
   520→
   521→/** טוויאפ-שקול-נוטיונל לזוג (שני-הכיוונים — base/quote או quote/base) */
   522→export function priceTruthFor(rt: ChainRT, base: string, quote: string): PriceTruth | null {
   523→  const st = rt.st;
   524→  let wSum = 0, pxSum = 0, n = 0, hMin = Infinity, hMax = 0;
   525→  for (const s of rt.priceSamples) {
   526→    const same = s.base === base && s.quote === quote;
   527→    const flip = s.base === quote && s.quote === base;
   528→    if (!same && !flip) continue;
   529→    const px = same ? s.px : Math.floor(1_000_000 / (s.px > 0 ? s.px : 1)); // יחס-הפוך: pxFlip = (1000/px)×1000 = 10^6/px
   530→    wSum += s.muQ; pxSum += px * s.muQ; n++;
   531→    if (s.h < hMin) hMin = s.h; if (s.h > hMax) hMax = s.h;
   532→  }
   533→  if (n === 0 || wSum <= 0) return null;
   534→  const twap = Math.floor(pxSum / wSum);
   535→  const ref = st.assets[base]?.refPrice ?? 0;
   536→  return { base, quote, twap, samples: n, devBpsVsRef: ref > 0 ? Math.floor(((twap - ref) / ref) * 10_000) : 0, windowHeights: hMax >= hMin ? hMax - hMin + 1 : 0 };
   537→}

   539→function ladderCheck(rt: ChainRT): void {
   540→  const st = rt.st;
   541→  if (rt.pendingLift) return; // FIX(Roast-65-a · MED-6): הרמה-קודמת עוד-לא-אושרה — אין-ערימת-כוונות
   542→  if (st.height - st.ladder.lastLiftHeight < LADDER_EVERY) return;
   543→  /* ── SETTLE-14 PRICE-TRUTH: הסולם-כמו-שדרה (keel), לא-מדרגות ──
   544→   * היה: newRef = ref×1.005 בלי-קשר-לשוק — מדרגות-עולות-בלבד (2500→2686
   545→   * מונוטוני, 15-הזזות). המפעיל צדק: "מטפס-ולא-נראה-כמו-נתון-אמת".
   546→   * עכשיו: הסולם זז רק-אחרי-עדות-מסחר-אמת (טוויאפ-מהטבעת), בשני-הכיוונים,
   547→   * מדומם (חצי-הפער, עד-0.5%), וממשיך-לשלם-עמלות-על-כל-הזזה (harvest).
   548→   * אין-עדות-מספקת (<40-מסחרות או |dev|<100bps) → הסולם לא-זז. */
   549→  const truth = priceTruthFor(rt, "SAOS", "USDS");
   550→  if (!truth || truth.samples < 40) return;
   551→  const ref = st.ladder.refPrice;
   552→  const devBps = truth.devBpsVsRef;
   553→  if (Math.abs(devBps) < 100) return;
   554→  if (st.feeMeter < st.ladder.stepCostMu) return; // עדיין-משלמים-בעמלות-אמת-על-כל-הזזה
   555→  const dir = devBps > 0 ? 1 : -1;
   556→  const halfDev = Math.floor(Math.abs(ref - truth.twap) / 2);
   557→  // מדומם-חצי-פער, עד-0.5%; פער-גולמי (|dev|>1000bps — מעגן-מיושן) מותר-עד-2%
   558→  // כדי-שהסולם-יתפוס-במהירות את-המציאות-שנצברה — באותה-מידה למעלה ולמטה.
   559→  const capPct = Math.abs(devBps) > 1_000 ? 0.02 : 0.005;
   560→  const step = Math.max(1, Math.min(Math.floor(ref * capPct), halfDev));
   561→  const newRef = Math.max(1, ref + dir * step);
   562→  const harvestMu = Math.min(st.ladder.stepCostMu, st.feeMeter);
   563→  const harvestNonce = sysNonce(st, "harv");
   564→  const priceNonce = sysNonce(st, "ladder");
   565→  rt.mempool.push({ type: "harvest", issuer: "treasury", nonce: harvestNonce, amount: harvestMu, memo: `keel-${dir > 0 ? "up" : "down"}` });
   566→  rt.mempool.push({ type: "asset_price", issuer: "treasury", nonce: priceNonce, asset: "SAOS", refPrice: newRef });
   567→  // FIX(Roast-65-a · MED-6): הספרים לא נוגעים-כאן-עוד. הישן מוטט את st.ladder בשעת-הדחיפה
   568→  // (refPrice/lastLiftHeight/lifts/spentMu/stats.ladders) לפני-שה-ops אושרו — אם-harvest
   569→  // נדחה (אוצר/מד-עמלות ירדו-בין-הדחיפה לביצוע) המצב העיד הרמה-שלא-שולמה וה-op-log לא
   570→  // משחזר אותה. עכשיו: כוונה-בלבד (rt.pendingLift) — הספרים זזים רק-אחרי-r.ok של שני-
   571→  // ה-ops (buildBlock), ודחייה מתועדת-כנה בלי-שינוי-מצב.
   572→  rt.pendingLift = { newRef, harvestMu, h: st.height, harvestNonce, priceNonce };
   573→  pushReceipt(rt, "keel-intent", { height: st.height, newRef, dir, devBps, step, harvestMu, twap: truth.twap, samples: truth.samples });
   574→}

   576→async function buildBlock(rt: ChainRT): Promise<void> {
   577→  const t0 = Date.now();
   578→  const st = rt.st;
   579→  const ops = rt.mempool.splice(0, 160);
   580→  let applied = 0;
   581→  let rejected = 0;
   582→  const rejSample: string[] = [];
   583→  // FIX(Roast-65-a · MED-6): מעקב-אישור-ה-ops-של-הרמה-הממתינה (harvest + asset_price)
   584→  let keelHarvest: boolean | null = null;
   585→  let keelPrice: boolean | null = null;
   586→  const trackKeel = (op: Op, ok: boolean): void => {
   587→    if (!rt.pendingLift) return;
   588→    if (op.nonce === rt.pendingLift.harvestNonce) keelHarvest = ok;
   589→    if (op.nonce === rt.pendingLift.priceNonce) keelPrice = ok;
   590→  };
   591→  // FIX(Roast-66 · dose-truth): מעקב-תוצאות-המנות-של-המסילה (נונס rail-*) — המנה
   592→  // נספרת-כבוצעה רק-מתוצאת-הבלוק-בפועל (היה: נספרה-בדחיפה, minOut-fail נספר-כמנה).
   593→  const railResults: { nonce: string; ok: boolean }[] = [];
   594→  const trackRail = (op: Op, ok: boolean): void => {
   595→    if (op.nonce.startsWith("rail-")) railResults.push({ nonce: op.nonce, ok });
   596→  };
   597→  // FIX(R17-b·H3): אותו-מעקב-ל-ops-הגשר (brg-*/blo-*) — paid/locked נסגרים-על-האמת
   598→  // (reject → היפוך-ל-pending/lock-failed + ניקוי-הרישום-הקבוע), לא-על-התור.
   599→  const bridgeResults: { nonce: string; ok: boolean }[] = [];
   600→  const trackBridge = (op: Op, ok: boolean): void => {
   601→    if (op.nonce.startsWith("brg-") || op.nonce.startsWith("blo-")) bridgeResults.push({ nonce: op.nonce, ok });
   602→  };
   603→  for (const op of ops) {
   604→    // FIX(Roast-65-a · MED-8): אטומיות-בלוק. הישן: זריקה-מ-applyOp באמצע-הלולאה הפילה
   605→    // את-כל-הבלוק אחרי-שה-ops-המוקדמים כבר שינו את st — מוטציות-חיות בלי blockHash/DB,
   606→    // 160-ה-ops נשפכו. עכשיו: try/catch-פר-op — כשל-לא-אינווריאנטי נרשם-כדחייה-כנה
   607→    // (אותו-נתיב של-r.err) והבלוק ממשיך-שלם. InvariantViolation עדיין-זורק (HALT-מוסדי).
   608→    let r: ReturnType<typeof applyOp>;
   609→    try {
   610→      r = applyOp(st, op, t0);
   611→    } catch (e) {
   612→      if (e instanceof InvariantViolation) throw e;
   613→      trackKeel(op, false);
   614→      trackRail(op, false);
   615→      trackBridge(op, false);
   616→      rejected++;
   617→      if (rejSample.length < 5) rejSample.push(`${op.type}(${(op as { issuer?: string }).issuer ?? "?"}):${String(e instanceof Error ? e.message : e).slice(0, 60)}`);
   618→      continue;
   619→    }
   620→    trackKeel(op, r.ok);
   621→    trackRail(op, r.ok);
   622→    trackBridge(op, r.ok);
   623→    if (r.ok) {
   624→      applied++;
   625→      for (const ev of r.events ?? []) rt.tape.unshift({ ...ev, h: st.height });
   626→      feedPriceOracle(rt, r.events, st.height); // SETTLE-14: עדות-מחיר-אמת מכל-מסחר
   627→    } else {
   628→      // FIX(SETTLE-15 silent-drop): דחייה-שקטה הייתה-חור-תפעולי-ישן ("queued" אך-
   629→      // נופלת-בבלוק-בלי-עדה). עכשיו: לוג-מדוד (דגימה-עד-5-פר-בלוק) — כל-דחייה
   630→      // נראית ב-dev.log עם-הסיבה, והמונה-החי-חשוף ב-getState.rejectedLast.
   631→      rejected++;
   632→      if (rejSample.length < 5) rejSample.push(`${op.type}(${(op as { issuer?: string }).issuer ?? "?"}):${r.err}`);
   633→    }
   634→  }
   635→  // FIX(Roast-66 · dose-truth): פיוס-המנות-של-המסילה אחרי-הבלוק (ספירה-בהצלחה-בלבד)
   636→  if (railResults.length > 0) railReconcile(railResults);
   637→  // FIX(R17-b·H3): פיוס-הגשר — סטטוסי-paid/locked נסגרים-מול-תוצאת-הבלוק-בפועל
   638→  if (bridgeResults.length > 0) bridgeReconcile(bridgeResults);
   639→  // FIX(Roast-65-a · MED-6): אישור-הספרים-רק-אחרי-ששני-ה-ops-של-הכוונה אכן היו-בבלוק
   640→  // ואושרו. דחייה = לוג-רועש + קבלה-כנה, אפס-שינוי-ב-st.ladder (הכוונה נזרקת).
   641→  if (rt.pendingLift && keelHarvest !== null && keelPrice !== null) {
   642→    const pl = rt.pendingLift;
   643→    rt.pendingLift = null;
   644→    if (keelHarvest && keelPrice) {
   645→      st.ladder.refPrice = pl.newRef;
   646→      st.ladder.lastLiftHeight = pl.h;
   647→      st.ladder.lifts++;
   648→      st.ladder.spentMu += pl.harvestMu;
   649→      st.stats.ladders++;
   650→      pushReceipt(rt, "keel", { height: pl.h, newRef: pl.newRef, harvestMu: pl.harvestMu, confirmed: true });
   651→    } else {
   652→      console.warn(`[saos-dex][ALERT] keel lift at h${pl.h} NOT CONFIRMED (harvest=${keelHarvest} price=${keelPrice}) — ladder books untouched (Roast-65-a MED-6: no unpaid lifts in state)`);
   653→      pushReceipt(rt, "keel-rejected", { height: pl.h, newRef: pl.newRef, harvestOk: keelHarvest, priceOk: keelPrice });
   654→    }
   655→  }
   656→  if (rejected > 0) {
   657→    rt.rejectedLast = { h: st.height, count: rejected, sample: rejSample };
   658→    console.warn(`[saos-dex] block h${st.height}: ${rejected}/${ops.length} ops rejected — ${rejSample.join(" · ")}`);
   659→  }
   660→  rt.tape = rt.tape.slice(0, TAPE_CAP);
   661→  // FIX(SETTLE-7 journal-bleed): st.journal הוא חלק-מה-DexState (ולכן מוטבע
   662→  // בתמונת-מצב ובשורש) — attest ×4/חותם + pay/refuel/mint/relay הצטברו-לנצח
   663→  // (זיכרון + דיסק + stateRootOf O(n) פעמיים-בבלוק). תקרה-קשה — ההיסטוריה
   664→  // המלאה חיה ב-evidenceReceipt/arenaJournal ב-DB, לא במצב-החי.
   665→  if (st.journal.length > 600) st.journal.splice(0, st.journal.length - 500);
   666→
   667→  // פעימה-אוטומטית (heartbeat — מדידת-חיות על-הספר)
   668→  applyOp(st, { type: "heartbeat", issuer: "treasury", nonce: sysNonce(st, "hb") }, t0);
   669→
   670→  // Task 54 · הטיק-הכלכלי הקנוני: revenue-router (קנייה-ושריפה) + אפוק-פליפ-ve —
   671→  // גובה-טריגר-טהור (אפס-שעון/רנדומליות) → ריפליי דטרמיניסטי.
   672→  // FIX(Roast-64): nowMs מוזרק — רשומות-הג'ורנל-של-הכלכלה נושאות את-זמן-הבלוק
   673→  // (היו t:0 — זמן-מזויף שנגל-את-חוב-ה-determinism-של-journal.t באופן-גרוע יותר).
   674→  economyTick(st, t0);
   675→
   676→  // Task 56 · מסחר-אוטומטי-חסר-קיפרים: DCA-פריסה + טריגרים (stop/take) —
   677→  // בונה-הבלוקים הוא ה-keper: הטיקים מבצעים applyOp ישירות-על-המצב (אטומי,
   678→  // סינכרוני, ללא-ממפול) והמסחר-שנוצר נכנס לטייפ ולעדות-המחיר-אמת.
   679→  // Task 58 · ProtoRev (ארב-סוף-בלוק) + CoW (סליקת-אצווה-מחיר-אחיד) —
   680→  // אותו-חוק: בונה-הבלוקים הוא ה-keeper/solver — אפס-תשתית-חוץ, ריפליי-אמת.
   681→  {
   682→    const exec = (op: Op) => applyOp(st, op, t0);
   683→    // FIX(Roast-65-a · LOW-13): nowMs מוזרק-גם-ל-DCA/טריגרים — רישומי-הג'ורנל שלהם
   684→    // היו t:0 (זמן-מזויף) בניגוד-לתיקון-63/64 על-שאר-הנתיבים.
   685→    // Task 80 · THE-VAULT: סוכן-הכספת אינו-סוחר — אין-לו-אירועי-טייפ (ריבית ≠ עמלת-סוואפ).
   686→    vaultTick(st, rt.tickCount, exec, t0);
   687→    // R40 · THE-SAVE: דבר-גורר-דבר — פירעון-כספת באותו-בלוק מזרים ריבית-אמת
   688→    // לחוסכים (אחרי-סוכן-הכספת, לפני-שאר-הטיקים — הזרימה נראית-בבלוק-עצמו).
   689→    savingsTick(st, t0);
   690→    saveDemoTick(st, rt.tickCount, exec, t0);
   691→    // R41 · THE-VE: דבר-גורר-דבר — נעילה ← הצבעה ← אפוק-פליפ מרוקן את-קרן-ה-ve
   692→    // לנועלים-החיים (בכל-אפוק, h%1200); האג"ח ממירה USDS-סוחר ל-SAOS-נעול — שתי
   693→    // החוליות-האחרונות של צינור-המפקיד: הפקדה ← זיכוי ← חיסכון ← הכנסה ← פדיון
   694→    // ← וגם: מסחר ← עמלות ← קרן ← נועלים. אותם-חוקים לכל-אחד.

   818→      srcs.push(`${asset}×${cross?.sources ?? "?"}`);
   819→    }
   820→    if (Object.keys(prices).length === 0) return;

   825→    // בממפול — קבוע-מהמצב (קביעות-ריפליי נשמרת — בלי-רנדומיות) ויחודי-לכל-דחיפה.

   996→function ladderAgg(st: DexState, base: string, quote: string, side: "buy" | "sell", levels = 12) {
   997→  const book = sortBook(st, base, quote, side);
   998→  const out: { price: number; total: number; orders: number }[] = [];
   999→  for (const o of book) {
  1000→    const rem = o.amount - o.filled;
  1001→    if (rem <= 0) continue;
  1002→    const last = out[out.length - 1];
  1003→    if (last && last.price === o.price) { last.total += rem; last.orders++; }
  1004→    else if (out.length < levels) out.push({ price: o.price, total: rem, orders: 1 });
  1005→    else break;
  1006→  }
  1007→  return out;
  1008→}
