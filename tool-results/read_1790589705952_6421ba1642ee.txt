   204→export type FuelFunder = (typeof FUEL_FUNDERS)[number];
   205→/** סדר-הסולם (SETTLE-5/9 · roast-79): arena → bot-a/b/c → treasury → operator.
   206→ * FIX(roast-79 · fuel-death-spiral): סדר-הקדימות הוא-התרופה. היה: ארנה→אוצר→מפעיל —
   207→ * נמדד-חי: שלושתם מתים (25.9k שאינו-עובר-מבחן-amt+10k · 8 · 2k) בזמן-שהבוטים
   208→ * מחזיקים 1.65M. עכשיו: ארנה → בוטים (מרווח לבריאות-המממן) → אוצר → מפעיל. */
   209→export const FUNDER_ORDER: FuelFunder[] = ["arena", "bot-a", "bot-b", "bot-c", "treasury", "operator"];
   210→/** FIX(roast-81 · afford-single-source): יכולת-המימון-המקסימלית של ממנה — אותם-מרווחי-
   211→ * ברזל של-funderFor (ארנה/בוטים שומרים-רצפת-10k לבריאות-המממן; האוצר/המפעיל ניתנים-עד-
   212→ * תום-היתרה). SETTLE-9 dedup: המרווחים חיים-פה-בלבד — הדריפט-פה = דליפה-כלכלית. */
   213→export const FUNDER_KEEP = 10_000;
   214→export function funderAfford(st: DexState, who: FuelFunder): number {
   215→  const cr = st.credits[who] ?? 0;
   216→  const keep = who === "treasury" || who === "operator" ? 0 : FUNDER_KEEP;
   217→  return Math.max(0, cr - keep);
   218→}
   219→/** ממנה-דלק מדורג (SETTLE-5/9 · roast-79 · roast-81-refactor-טהור):
   220→ * exclude — לא-לממן-עצמי (הארנה אסור-שתממן-את-עצמה: המנוע דוחה rTo==issuer).
   221→ * התנהגות-זהה-במדויק לקודם — רק-המרווחים עברו-למקור-אמת-אחד (funderAfford). */
   222→export function funderFor(st: DexState, amt: number, exclude?: string): FuelFunder {
   223→  for (const f of FUNDER_ORDER) {
   224→    if (f === exclude) continue;
   225→    if (funderAfford(st, f) >= amt) return f;
   226→  }
   227→  return "operator";
   228→}

   634→/** ערך בטחונות ב-µUSDS: refPrice ב-mµ-ליחידה (USDS=1000 → $1; WSTEEM=54 → $0.054). */

   639→/** ריבית מתומחרת-במצב-השרשרת: בסיס 5%/שנה + תוספת-ניצולת (חוב-גלובלי/תקרה). */
   640→export function vaultRateBps(st: DexState): number {
   641→  let debt = 0;
   642→  for (const v of Object.values(st.vaults ?? {})) debt += v.debtMu + v.intMu; // קרן + ריבית-מונפת — התביעה-הכוללת
   643→  const util = debt >= VAULT_GLOBAL_CAP_MU ? 1 : debt / VAULT_GLOBAL_CAP_MU;
   644→  return VAULT_BASE_RATE_BPS + Math.floor(util * VAULT_UTIL_SURGE_BPS);
   645→}
   646→/** חוב-אפקטיבי = קרן + ריבית שנצברה — הונפה-עצלה: פונקציה-טהורה-של-המצב, אפס-כתיבה-פר-בלוק. */
   647→export function vaultEffDebtMu(debtMu: number, rateBps: number, fromH: number, toH: number): number {
   648→  const blocks = Math.max(0, toH - fromH);
   649→  if (blocks <= 0 || debtMu <= 0) return debtMu;
   650→  return debtMu + Math.floor((debtMu * rateBps * blocks) / (10_000 * BLOCKS_PER_YEAR));
   651→}

   677→  return (st.vaultTot?.interestMu ?? 0)
   678→    + (st.lend?.totals?.resToTreasuryMu ?? 0)

   682→/* ── R44 · THE-MARKET — מתמטיקה-טהורה של שוק-הכספים (שלמים-בלבד, דטרמיניסטי) ── */
   683→/** מצב-שוק — תמיד-אובייקט (תמונות-ישנות = שוק-ריק-כנה, לא-התרסקות). */
   684→export function lendState(st: DexState): NonNullable<DexState["lend"]> {
   685→  return st.lend ?? { markets: {}, positions: {}, totals: { suppliesMu: 0, borrowsMu: 0, interestPaidMu: 0, resToTreasuryMu: 0, liquidations: 0, opens: 0, closes: 0 } };
   686→}
   687→/** ניצולת-השוק ב-bps: חוב-קרן / אספקה-קרן. אספקה-אפסית → 0 (אין-שוק). */
   688→export function lendUtilBps(m: { sMu: number; bMu: number } | undefined): number {
   689→  if (!m || m.sMu <= 0) return 0;
   690→  return Math.min(10_000, Math.floor((m.bMu * 10_000) / m.sMu));
   691→}
   692→/** ריבית-שאילה-שנתית ב-bps — עקומת-kink מלאה (Aave-v3 standard). */
   693→export function lendBorrowAprBps(m: { sMu: number; bMu: number } | undefined): number {
   694→  const u = lendUtilBps(m);
   695→  if (u <= 0) return LEND_BASE_APR_BPS;
   696→  if (u <= LEND_KINK_BPS) return LEND_BASE_APR_BPS + Math.floor((LEND_SLOPE1_BPS * u) / LEND_KINK_BPS);
   697→  return LEND_BASE_APR_BPS + LEND_SLOPE1_BPS + Math.floor((LEND_SLOPE2_BPS * (u - LEND_KINK_BPS)) / (10_000 - LEND_KINK_BPS));
   698→}
   699→/** ריבית-אספקה-שנתית ב-bps — שאילה × ניצולת × (1 − פקטור-רזרבה). */
   700→export function lendSupplyAprBps(m: { sMu: number; bMu: number } | undefined): number {
   701→  const u = lendUtilBps(m);
   702→  if (u <= 0) return 0;
   703→  return Math.floor((lendBorrowAprBps(m) * u * (10_000 - LEND_RESERVE_FACTOR_BPS)) / 100_000_000);
   704→}
   705→/** חוב-אפקטיבי = קרן + ריבית-מונפת + צבירה-טרייה — פונקציה-טהורה-של-הגובה. */
   706→export function lendDebtNow(b: { mu: number; int: number; atH: number }, aprBps: number, h: number): number {
   707→  const base = b.mu + b.int;
   708→  const blocks = Math.max(0, h - b.atH);
   709→  if (base <= 0 || blocks <= 0) return base;
   710→  return base + Math.floor((base * aprBps * blocks) / (10_000 * BLOCKS_PER_YEAR));
   711→}
   712→/** ערך-נכס ב-µUSDS לפי-האורקל-הקנוני (אותו-סולם-מדויק של הכספת: refPrice=mµ/יחידה). */
   713→export function lendAssetValueMu(st: DexState, asset: string, mu: number): number {
   714→  const ref = st.assets[asset]?.refPrice ?? 0;
   715→  return ref > 0 ? Math.floor((mu * ref) / 1000) : 0;
   716→}
   717→/** HF×100 = בטחונות(בסף-הניזולת) / חוב. אין-חוב → 1e9 (אינסוף-מיוצג). */
   718→export function lendHealthFactor(st: DexState, who: string): number {
   719→  const ls = lendState(st);
   720→  const pos = ls.positions[who];
   721→  if (!pos) return 1e9;
   722→  let col = 0, debt = 0;
   723→  for (const [a, sp] of Object.entries(pos.s)) {
   724→    col += Math.floor((lendAssetValueMu(st, a, sp.mu + sp.earned) * lendLiqThresholdBps(a)) / 10_000);
   725→  }
   726→  for (const [a, bp] of Object.entries(pos.b)) {
   727→    if (bp.mu + bp.int <= 0) continue;
   728→    const debtMu = lendDebtNow(bp, lendBorrowAprBps(ls.markets[a]), st.height);
   729→    debt += lendAssetValueMu(st, a, debtMu);
   730→  }
   731→  if (debt <= 0) return 1e9;
   732→  return Math.floor((col * 100) / debt);
   733→}

   735→export function applyOp(st: DexState, op: Op, nowMs: number): OpResult {
   736→  if (!op || typeof op !== "object") return { ok: false, err: "bad-op", gas: 0 };
   737→  const gas = GAS_TABLE[op.type];
   738→  if (gas === undefined) return { ok: false, err: "unknown-op", gas: 0 };
   739→  // FIX(R15-a · CRITICAL-1 · integer-gates): op.amount גולמי (100.5) עבר כמה-שערים
   740→  // (refuel/mint_fuel/wrap) שכתבו אותו ישר-למצב-הקנוני → stateRoot זרק CanonicalError
   741→  // → HALT, וההתאוששות היחידה היא reset שמוחק-ראיות. עתה: שער-שלמים אחיד בדלת-ה-op —
   742→  // נדחה לפני-כל-נגיעה-במצב (אותו-עיקרון Roast-65-a HIGH-1 שהוחל-רק על order/cow/price).
   743→  if (op.amount !== undefined && !Number.isSafeInteger(op.amount))
   744→    return { ok: false, err: "non-integer-amount", gas: 0 };
   745→  const issuer = String(op.issuer ?? "").toLowerCase();
   746→  if (!issuer || !op.nonce) return { ok: false, err: "issuer+nonce-required", gas: 0 };
   747→  // FIX(Roast-64 · prototype-pollution): זיהוי-מסוכן נדחה-בדלת (שים-לב: גם addBal/
   748→  // addLock מוגנים — הגנה-כפולה fail-closed). נמען-"__proto__" נדחה בדיוק-כך.
   749→  if (FORBIDDEN_KEYS.has(issuer)) return { ok: false, err: "forbidden-issuer", gas: 0 };
   750→  // FIX(SETTLE-7 authz-split): נרמול-פעם-אחת בכניסה — כל-הנתיבים (יתרות/נעילות/
   751→  // order-cap/ספר/בריכות) חייבים-לראות את אותו-מזהה. החור-ההיסטורי: issuer-גולמי
   752→  // ("TREASURY") עבר בדיקת-funder-only במנוע-המנורמל אך נגע ביתרות-של-מזהה-אחר.
   753→  op.issuer = issuer;
   754→  if (typeof op.to === "string") {
   755→    op.to = op.to.toLowerCase(); // אותו-עיקרון לנמענים — אין-חשבונות-רפאים בגלל-רישוי
   756→    if (FORBIDDEN_KEYS.has(op.to)) return { ok: false, err: "forbidden-recipient", gas: 0 };
   757→  }
   758→  const dedupeKey = `${issuer}:${op.nonce}`;
   759→  if (nonceSet(st).has(dedupeKey)) return { ok: false, err: "replay-rejected", gas: 0 };
   760→
   761→  const consume = (): OpResult | null => {
   762→    if (!chargeGas(st, issuer, gas)) return { ok: false, err: "insufficient-credits", gas };
   763→    pushNonce(st, dedupeKey);
   764→    st.stats.opsTotal++;
   765→    return null;
   766→  };
   767→
   768→  switch (op.type) {
   769→    case "ve_lock": {
   770→      // Task 54 · veSAOS (Aerodrome ve(3,3)): SAOS→כספת-ve, עמדה-אחת-פר-חשבון;
   771→      // הצטברות = מיזוג-עמדה + הארכת-unlock למקסימום. כוח-ve דועך-לינארית.
   772→      const a = op.asset, amt = op.amount ?? 0, weeks = Math.floor(op.weeks ?? 0);
   773→      if (a !== "SAOS") return { ok: false, err: "ve-saos-only", gas: 0 };
   774→      if (!(amt > 0)) return { ok: false, err: "bad-amount", gas: 0 };
   775→      if (!Number.isInteger(weeks) || weeks < VE_MIN_WEEKS || weeks > VE_MAX_WEEKS)
   776→        return { ok: false, err: "bad-weeks", gas: 0 };
   777→      if (avail(st, issuer, "SAOS") < amt) return { ok: false, err: "insufficient", gas };
   778→      const c = consume(); if (c) return c;
   779→      addBal(st, issuer, "SAOS", -amt);
   780→      addBal(st, VE_VAULT, "SAOS", amt);
   781→      const prev = st.ve?.[issuer];
   782→      const unlockFromWeeks = st.height + weeks * BLOCKS_PER_WEEK;
   783→      st.ve = {
   784→        ...(st.ve ?? {}),
   785→        [issuer]: {
   786→          amt: (prev?.amt ?? 0) + amt,
   787→          unlockH: Math.max(prev?.unlockH ?? 0, unlockFromWeeks),
   788→          weeks: Math.max(prev?.weeks ?? 0, weeks),
   789→          atH: st.height,
   790→        },
   791→      };
   792→      st.journal.push({ h: st.height, t: nowMs, soldier: issuer, action: "ve_lock", note: `${amt}µ SAOS × ${weeks}w → ve` });
   793→      return { ok: true, gas };
   794→    }
   795→    case "ve_unlock": {
   796→      // פריקה — רק-אחרי-תום-הנעילה (fail-closed: הגובה הוא-החוק)
   797→      const pos = st.ve?.[issuer];
   798→      if (!pos) return { ok: false, err: "no-ve-position", gas: 0 };
   799→      if (st.height < pos.unlockH) return { ok: false, err: "ve-still-locked", gas: 0 };
   800→      const c = consume(); if (c) return c;
   801→      addBal(st, VE_VAULT, "SAOS", -pos.amt);
   802→      addBal(st, issuer, "SAOS", pos.amt);
   803→      const votes = { ...(st.veVotes ?? {}) }; delete votes[issuer];
   804→      st.veVotes = votes;
   805→      const veMap = { ...(st.ve ?? {}) }; delete veMap[issuer];
   806→      st.ve = veMap;
   807→      st.journal.push({ h: st.height, t: nowMs, soldier: issuer, action: "ve_unlock", note: `${pos.amt}µ SAOS → free` });
   808→      return { ok: true, gas };
   809→    }
   810→    case "ve_vote": {
   811→      // הצבעת-גייג': קרן-התגמולים זורמת-לבריכות-שמוצבות (Aerodrome: 100%-דמי-הסוואפ
   812→      // למצביעים). משקל = כוח-ve-בשעת-הפליפ; הצבעה-חדשה דורסת-ישנה (עדכון-חופשי).
   813→      const pair = String(op.pair ?? "");
   814→      if (!st.pools[pair]) return { ok: false, err: "pool-not-found", gas: 0 };
   815→      if (!st.ve?.[issuer] || st.ve[issuer].amt <= 0) return { ok: false, err: "no-ve-position", gas: 0 };
   816→      const c = consume(); if (c) return c;
   817→      st.veVotes = { ...(st.veVotes ?? {}), [issuer]: pair };
   818→      st.journal.push({ h: st.height, t: nowMs, soldier: issuer, action: "ve_vote", note: `→ ${pair}` });
   819→      return { ok: true, gas };
   820→    }
   821→    case "bond_buy": {
   822→      // Task 54 · BONDS-2.0 (Olympus-POL, נלמד-מהכשל): SAOS-אוצר בהנחת-3% מול-USDS;
   823→      // התמורה יורדת-לאוצר, ה-SAOS נכנס-ve-נעול-12שב' לקונה — נזילות-אוצר
   824→      // שאי-אפשר-למשוך + אפס-הדפסה (נמכר-רק-מהיתרה-האמת של-האוצר).
   825→      const usdsIn = Math.floor(op.amount ?? 0);
   826→      if (!(usdsIn >= BOND_MIN_USDS && usdsIn <= BOND_MAX_USDS)) return { ok: false, err: "bond-size", gas: 0 };
   827→      if (avail(st, issuer, "USDS") < usdsIn) return { ok: false, err: "insufficient", gas };
   828→      const last = st.bondUserH?.[issuer] ?? 0;
   829→      if (st.height - last < BOND_COOLDOWN) return { ok: false, err: "bond-cooldown", gas: 0 };
   830→      const pool = st.pools["SAOS/USDS"];
   831→      if (!pool || pool.ra <= 0) return { ok: false, err: "no-ref-pool", gas: 0 };
   832→      const price = Math.floor((pool.rb * 1000) / pool.ra); // µ-USDS ל-1µ-SAOS (×1000-סקאלה)
   833→      if (!(price > 0)) return { ok: false, err: "no-price", gas: 0 };
   834→      const saosOut = Math.floor((usdsIn * 1000 * 10_000) / (price * (10_000 - BOND_DISC_BPS)));
   835→      if (!(saosOut > 0)) return { ok: false, err: "bad-quote", gas: 0 };
   836→      if (bal(st, "treasury", "SAOS") < saosOut) return { ok: false, err: "treasury-empty", gas: 0 };
   837→      const c = consume(); if (c) return c;
   838→      addBal(st, issuer, "USDS", -usdsIn);
   839→      addBal(st, "treasury", "USDS", usdsIn);
   840→      addBal(st, "treasury", "SAOS", -saosOut);
   841→      addBal(st, VE_VAULT, "SAOS", saosOut);
   842→      const prev = st.ve?.[issuer];
   843→      const unlockFromWeeks = st.height + BOND_VE_WEEKS * BLOCKS_PER_WEEK;
   844→      st.ve = {
   845→        ...(st.ve ?? {}),
   846→        [issuer]: {
   847→          amt: (prev?.amt ?? 0) + saosOut,
   848→          unlockH: Math.max(prev?.unlockH ?? 0, unlockFromWeeks),
   849→          weeks: Math.max(prev?.weeks ?? 0, BOND_VE_WEEKS),
   850→          atH: st.height,
   851→        },
   852→      };
   853→      st.bond = {
   854→        soldSaosMu: (st.bond?.soldSaosMu ?? 0) + saosOut,
   855→        usdsInMu: (st.bond?.usdsInMu ?? 0) + usdsIn,
   856→        count: (st.bond?.count ?? 0) + 1,
   857→      };
   858→      st.bondUserH = { ...(st.bondUserH ?? {}), [issuer]: st.height };
   859→      st.journal.push({ h: st.height, t: nowMs, soldier: issuer, action: "bond_buy", note: `${usdsIn}µ USDS → ${saosOut}µ SAOS (-3%, ve ${BOND_VE_WEEKS}w)` });
   860→      return { ok: true, gas };

  1759→        totals: { ...ls0.totals, suppliesMu: ls0.totals.suppliesMu + amt, opens: ls0.totals.opens + 1 },
  1760→      };
  1761→      st.lend = ls;
  1762→      st.journal.push({ h: st.height, t: nowMs, soldier: issuer, action: "lend_supply", note: `+${amt}µ ${asset} → lend-pot (pool ${(m0?.sMu ?? 0) + amt}µ · util ${lendUtilBps(ls.markets[asset]) / 100}% · borrowAPR ${lendBorrowAprBps(ls.markets[asset]) / 100}%)` });
  1763→      return { ok: true, gas };
  1764→    }
  1765→    case "lend_withdraw": {
  1766→      // משיכת-קרן + כל-הריבית-המדודה-שנצברה. כנות-נזילות: אם-הבריכה-מרוקנת
  1767→      // (הלווים מחזיקים) — pot-short כנה, כמו-החיסכון. רצפת-ה-10% בשאילה
  1768→      // מבטיחה-שתמיד-תהיה-משיכה-אפשרית-בחלק-מהקרן.
  1769→      const asset = String(op.asset ?? "");
  1770→      if (!(asset in LEND_MARKETS)) return { ok: false, err: "market-not-listed", gas: 0 };
  1771→      const ls = lendState(st);
  1772→      const pos = ls.positions[issuer];
  1773→      const sp = pos?.s[asset];
  1774→      if (!pos || !sp || sp.mu + sp.earned <= 0) return { ok: false, err: "no-position", gas: 0 };
  1775→      const amt = Math.floor(op.amount ?? -1); // השמטה/-1 = הכל
  1776→      if (amt === 0) return { ok: false, err: "bad-amount", gas: 0 };
  1777→      const takeMu = amt < 0 ? sp.mu : Math.min(amt, sp.mu);
  1778→      const earned = sp.earned; // ריבית-מדודה משולמת-תמיד-במלואה (מדפוס-החיסכון)
  1779→      if (avail(st, LEND_POT, asset) < takeMu + earned) return { ok: false, err: "pot-short", gas };
  1780→      const c = consume(); if (c) return c;
  1781→      addBal(st, LEND_POT, asset, -(takeMu + earned));
  1782→      addBal(st, issuer, asset, takeMu + earned);
  1783→      const m = ls.markets[asset]!;
  1784→      const restMu = sp.mu - takeMu;
  1785→      const sMaps = { ...pos.s };
  1786→      if (restMu <= 0 && earned <= 0) delete sMaps[asset]; else sMaps[asset] = { mu: restMu, earned: 0 };
  1787→      const pMaps = { ...ls.positions };
  1788→      if (Object.keys(sMaps).length === 0 && Object.keys(pos.b).length === 0) delete pMaps[issuer]; else pMaps[issuer] = { ...pos, s: sMaps };
  1789→      st.lend = {
  1790→        ...ls,
  1791→        markets: { ...ls.markets, [asset]: { ...m, sMu: m.sMu - takeMu, sEarnedMu: Math.max(0, m.sEarnedMu - earned), lastH: st.height } },
  1792→        positions: pMaps,
  1793→        totals: { ...ls.totals, suppliesMu: ls.totals.suppliesMu - takeMu, closes: ls.totals.closes + 1 },
  1794→      };
  1795→      st.journal.push({ h: st.height, t: nowMs, soldier: issuer, action: "lend_withdraw", note: `withdraw ${takeMu}µ ${asset} + earned ${earned}µ (measured) · pool ${m.sMu - takeMu}µ` });
  1796→      return { ok: true, gas };
  1797→    }
  1798→    case "lend_borrow": {
  1799→      // שאילה-מהבריכה נגד-בטחונות: HF ≥ 1.10 (חוצץ), רצפת-נזילות-10% (המלווים
  1800→      // תמיד-יוכלו-למשוך), תקרות-שוק-ועמדה. החוב נפתח-בקרן-נקייה + ריבית-מונפת
  1801→      // בגובה (אותה-מתמטיקה-דטרמיניסטית-של-הכספת — צבירה-עצלה-בקריאה).
  1802→      const asset = String(op.asset ?? "");
  1803→      if (!(asset in LEND_MARKETS)) return { ok: false, err: "market-not-listed", gas: 0 };
  1804→      const amt = Math.floor(op.amount ?? 0);
  1805→      if (!Number.isSafeInteger(amt) || amt <= 0) return { ok: false, err: "bad-amount", gas: 0 };
  1806→      const ls = lendState(st);
  1807→      const m = ls.markets[asset];
  1808→      if ((m?.bMu ?? 0) + amt > LEND_GLOBAL_CAP_MU) return { ok: false, err: "market-cap", gas: 0 };
  1809→      const pos = ls.positions[issuer] ?? { s: {}, b: {} };
  1810→      // תקציב-נזילות: השאילה לא-תרוקן-את-הבריכה מתחת-לרצפת-10% מהאספקה
  1811→      const cash = avail(st, LEND_POT, asset);
  1812→      const floorMu = Math.floor(((m?.sMu ?? 0) * (10_000 - LEND_MAX_UTIL_BPS)) / 10_000);
  1813→      if (cash - amt < floorMu) return { ok: false, err: "cash-floor", gas: 0 };
  1814→      if (cash < amt) return { ok: false, err: "pot-short", gas: 0 };
  1815→      // בדיקת-בריאות-מלאה: אחרי-השאילה HF (ב-LTV) ≥ 1.10
  1816→      const simB = { ...pos.b };
  1817→      const prevB = simB[asset];
  1818→      if (prevB) {
  1819→        const debtNow = lendDebtNow(prevB, lendBorrowAprBps(m), st.height);
  1820→        simB[asset] = { mu: prevB.mu + amt, int: debtNow - prevB.mu, atH: st.height };
  1821→      } else {
  1822→        simB[asset] = { mu: amt, int: 0, atH: st.height };
  1823→      }
  1824→      let colV = 0, debtV = 0;
  1825→      for (const [a, s2] of Object.entries(pos.s)) {
  1826→        const ltv = LEND_MARKETS[a] ?? 0;
  1827→        colV += Math.floor((lendAssetValueMu(st, a, s2.mu + s2.earned) * ltv) / 10_000);
  1828→      }
  1829→      for (const [a, b2] of Object.entries(simB)) {
  1830→        if (b2.mu + b2.int <= 0) continue;
  1831→        const d = lendDebtNow(b2, lendBorrowAprBps(ls.markets[a]), st.height);
  1832→        debtV += lendAssetValueMu(st, a, d);
  1833→      }
  1834→      if (debtV > colV || colV <= 0) return { ok: false, err: "no-collateral", gas: 0 };
  1835→      if (Math.floor((colV * 100) / debtV) < LEND_HF_BUFFER) return { ok: false, err: "health-factor", gas: 0 };
  1836→      const c = consume(); if (c) return c;
  1837→      addBal(st, LEND_POT, asset, -amt);
  1838→      addBal(st, issuer, asset, amt);
  1839→      const bMaps = { ...pos.b };
  1840→      if (prevB) {
  1841→        const debtNow = lendDebtNow(prevB, lendBorrowAprBps(m), st.height);
  1842→        bMaps[asset] = { mu: prevB.mu + amt, int: debtNow - prevB.mu, atH: st.height };
  1843→      } else {
  1844→        bMaps[asset] = { mu: amt, int: 0, atH: st.height };
  1845→      }
  1846→      st.lend = {
  1847→        ...ls,
  1848→        markets: { ...ls.markets, [asset]: { ...(m ?? { sMu: 0, bMu: 0, sEarnedMu: 0, resMu: 0, lastH: 0 }), bMu: (m?.bMu ?? 0) + amt, lastH: st.height } },
  1849→        positions: { ...ls.positions, [issuer]: { ...pos, b: bMaps } },
  1850→        totals: { ...ls.totals, borrowsMu: ls.totals.borrowsMu + amt },
  1851→      };
  1852→      st.journal.push({ h: st.height, t: nowMs, soldier: issuer, action: "lend_borrow", note: `borrow ${amt}µ ${asset} (util ${lendUtilBps(st.lend.markets[asset]) / 100}% · APR ${lendBorrowAprBps(st.lend.markets[asset]) / 100}% · HF after ${Math.floor((colV * 100) / debtV) / 100})` });
  1853→      return { ok: true, gas };
  1854→    }
  1855→    case "lend_repay": {
  1856→      // פירעון — הנה מגיע-הכסף-האמת: הריבית שהלווה-משלם (µ-אמת-מיתרתו) נחתכת:
  1857→      // 90% → מלווים פרו-ראטה-לפי-קרן (מדידה-חיה, אחרון-בולע-אבק), 10% → אוצר.
  1858→      // אפס-הטבעה: הכל-העברות-טהורות. אינווריאנטת-הספרים נשמרת-מעצמה.
  1859→      const asset = String(op.asset ?? "");
  1860→      if (!(asset in LEND_MARKETS)) return { ok: false, err: "market-not-listed", gas: 0 };
  1861→      const ls = lendState(st);
  1862→      const pos = ls.positions[issuer];
  1863→      const bp = pos?.b[asset];
  1864→      if (!pos || !bp || bp.mu + bp.int <= 0) return { ok: false, err: "no-debt", gas: 0 };
  1865→      const amt = Math.floor(op.amount ?? -1); // השמטה/-1 = הכל
  1866→      if (amt === 0) return { ok: false, err: "bad-amount", gas: 0 };
  1867→      const m = ls.markets[asset]!;
  1868→      const debt = lendDebtNow(bp, lendBorrowAprBps(m), st.height);
  1869→      const pay = amt < 0 ? debt : Math.min(amt, debt);
  1870→      if (pay <= 0) return { ok: false, err: "nothing-to-do", gas: 0 };
  1871→      if (avail(st, issuer, asset) < pay) return { ok: false, err: "insufficient-balance", gas };
  1872→      const accrued = debt - bp.mu;                     // כל-הריבית-שנצברה (מונפת + טרייה)
  1873→      const interestPart = Math.min(pay, accrued);      // ריבית-נגבית-קודם
  1874→      const principalPart = pay - interestPart;
  1875→      const c = consume(); if (c) return c;
  1876→      addBal(st, issuer, asset, -pay);
  1877→      addBal(st, LEND_POT, asset, pay);
  1878→      // חלוקת-הריבית: 90% מלווים / 10% אוצר — מהכסף-שהגיע-בפועל
  1879→      const toSuppliers = Math.floor((interestPart * (10_000 - LEND_RESERVE_FACTOR_BPS)) / 10_000);
  1880→      const toReserve = interestPart - toSuppliers;
  1881→      if (toReserve > 0) addBal(st, LEND_POT, asset, -toReserve), addBal(st, "treasury", asset, toReserve);
  1882→      // פרו-ראטה-לפי-קרן-חיה (האחרון-בסדר-הקנוני בולע-את-האבק — שימור-מדויק-עד-ה-µ)
  1883→      const positions = { ...ls.positions };
  1884→      let left = toSuppliers;
  1885→      const suppliers = Object.keys(positions).filter((k) => (positions[k]!.s[asset]?.mu ?? 0) > 0);
  1886→      for (let i = 0; i < suppliers.length; i++) {
  1887→        const p = positions[suppliers[i]]!;
  1888→        const sp = p.s[asset]!;
  1889→        if (i === suppliers.length - 1) {
  1890→          if (left > 0) { positions[suppliers[i]] = { ...p, s: { ...p.s, [asset]: { ...sp, earned: sp.earned + left } } }; left = 0; }
  1891→        } else {
  1892→          const share = Math.floor((toSuppliers * sp.mu) / (m.sMu));
  1893→          if (share > 0) { positions[suppliers[i]] = { ...p, s: { ...p.s, [asset]: { ...sp, earned: sp.earned + share } } }; left -= share; }
  1894→        }
  1895→      }
  1896→      // סגירת-החוב: ריבית-קודם, קרן-אחר-כך (אותו-סדר-חשבונאי-כמו-הכספת)
  1897→      const bMaps = { ...positions[issuer]!.b };
  1898→      const newInt = accrued - interestPart;
  1899→      const newMu = bp.mu - principalPart;
  1900→      if (newMu + newInt <= 0) delete bMaps[asset]; else bMaps[asset] = { mu: newMu, int: newInt, atH: st.height };
  1901→      const pAfter = positions[issuer]!;
  1902→      if (Object.keys(bMaps).length === 0 && Object.keys(pAfter.s).length === 0) delete positions[issuer]; else positions[issuer] = { ...pAfter, b: bMaps };
  1903→      st.lend = {
  1904→        ...ls,
  1905→        markets: { ...ls.markets, [asset]: { ...m, bMu: Math.max(0, m.bMu - principalPart), sEarnedMu: m.sEarnedMu + toSuppliers, resMu: m.resMu + toReserve, lastH: st.height } },
  1906→        positions,
  1907→        totals: { ...ls.totals, borrowsMu: Math.max(0, ls.totals.borrowsMu - principalPart), interestPaidMu: ls.totals.interestPaidMu + interestPart, resToTreasuryMu: ls.totals.resToTreasuryMu + toReserve },
  1908→      };
  1909→      st.journal.push({ h: st.height, t: nowMs, soldier: issuer, action: "lend_repay", note: `repay ${pay}µ ${asset} (interest ${interestPart}µ: ${toSuppliers}µ→${suppliers.length} suppliers · ${toReserve}µ→treasury · principal ${principalPart}µ) · debt now ${newMu + newInt}µ` });
  1910→      return { ok: true, gas };
  1911→    }
  1912→    case "lend_liquidate": {
  1913→      // ניזולת-אזרח: HF < 1 — כל-אחד-יכול-לסגור-חוב-ולקחת-בטחונות+5%.
  1914→      // הבטחון-הנתפס: הנכס-המסופק-בעל-הערך-הגדול-של-הלווה (דטרמיניסטי).
  1915→      const owner = String(op.owner ?? "").toLowerCase();
  1916→      if (!owner || FORBIDDEN_KEYS.has(owner)) return { ok: false, err: "bad-owner", gas: 0 };
  1917→      const asset = String(op.asset ?? "");
  1918→      if (!(asset in LEND_MARKETS)) return { ok: false, err: "market-not-listed", gas: 0 };
  1919→      const ls = lendState(st);
  1920→      const pos = ls.positions[owner];
  1921→      const bp = pos?.b[asset];
  1922→      if (!pos || !bp || bp.mu + bp.int <= 0) return { ok: false, err: "no-debt", gas: 0 };
  1923→      if (lendHealthFactor(st, owner) >= 100) return { ok: false, err: "not-liquidatable", gas: 0 };
  1924→      const m = ls.markets[asset]!;
  1925→      const debt = lendDebtNow(bp, lendBorrowAprBps(m), st.height);
  1926→      const pay = Math.min(Math.floor(op.amount ?? -1) < 0 ? debt : Math.floor(op.amount ?? -1), debt);
  1927→      if (pay <= 0) return { ok: false, err: "bad-amount", gas: 0 };
  1928→      if (avail(st, issuer, asset) < pay) return { ok: false, err: "insufficient-balance", gas };
  1929→      // בטחון-היעד: הספקה-הגדולה-בערך (שוויון → סדר-אלפביתי — דטרמיניסטי)
  1930→      let colAsset = "", colVal = 0;
  1931→      for (const [a, sp] of Object.entries(pos.s)) {
  1932→        const v = lendAssetValueMu(st, a, sp.mu + sp.earned);
  1933→        if (v > colVal) { colVal = v; colAsset = a; }
  1934→      }
  1935→      if (!colAsset || colVal <= 0) return { ok: false, err: "no-collateral", gas: 0 };
  1936→      const ref = st.assets[colAsset]?.refPrice ?? 0;
  1937→      if (ref <= 0) return { ok: false, err: "no-collateral-price", gas: 0 };
  1938→      const seizeValMu = Math.min(Math.floor((pay * (10_000 + LEND_LIQ_BONUS_BPS)) / 10_000), colVal);
  1939→      const colSp = pos.s[colAsset]!;
  1940→      const seizeMu = Math.min(colSp.mu + colSp.earned, Math.floor((seizeValMu * 1000) / ref));
  1941→      if (seizeMu <= 0) return { ok: false, err: "seize-too-small", gas: 0 };
  1942→      if (avail(st, LEND_POT, colAsset) < seizeMu) return { ok: false, err: "pot-short", gas: 0 };
  1943→      const c = consume(); if (c) return c;
  1944→      addBal(st, issuer, asset, -pay);
  1945→      addBal(st, LEND_POT, asset, pay);
  1946→      const accrued = debt - bp.mu;
  1947→      const interestPart = Math.min(pay, accrued);
  1948→      const principalPart = pay - interestPart;
  1949→      const toSuppliers = Math.floor((interestPart * (10_000 - LEND_RESERVE_FACTOR_BPS)) / 10_000);
  1950→      const toReserve = interestPart - toSuppliers;
  1951→      if (toReserve > 0) addBal(st, LEND_POT, asset, -toReserve), addBal(st, "treasury", asset, toReserve);
  1952→      const positions = { ...ls.positions };
  1953→      let left = toSuppliers;
  1954→      const suppliers = Object.keys(positions).filter((k) => (positions[k]!.s[asset]?.mu ?? 0) > 0);
  1955→      for (let i = 0; i < suppliers.length; i++) {
  1956→        const p = positions[suppliers[i]]!;
  1957→        const sp = p.s[asset]!;
  1958→        if (i === suppliers.length - 1) {
  1959→          if (left > 0) { positions[suppliers[i]] = { ...p, s: { ...p.s, [asset]: { ...sp, earned: sp.earned + left } } }; left = 0; }
  1960→        } else {
  1961→          const share = Math.floor((toSuppliers * sp.mu) / m.sMu);
  1962→          if (share > 0) { positions[suppliers[i]] = { ...p, s: { ...p.s, [asset]: { ...sp, earned: sp.earned + share } } }; left -= share; }
  1963→        }
  1964→      }
  1965→      // תפיסת-הבטחון: מה-pot אל-המפרק (הבטחון-הופקד-שם-פיזית)
  1966→      addBal(st, LEND_POT, colAsset, -seizeMu);
  1967→      addBal(st, issuer, colAsset, seizeMu);
  1968→      // חלוקת-התפיסה בין-קרן-לריבית (שתי-המראות חייבות-להתאים-לעמדות):
  1969→      // takeFromEarned נגבע-קודם — הריבית-שנצברה שייכת-לבעלים באותה-מידה
  1970→      const oPos = positions[owner]!;
  1971→      const oS = { ...oPos.s };
  1972→      const seizedSp = oS[colAsset]!;
  1973→      const takeFromEarned = Math.min(seizeMu, seizedSp.earned);
  1974→      const takeFromMu = seizeMu - takeFromEarned;
  1975→      const restMu = seizedSp.mu - takeFromMu;
  1976→      const restEarned = seizedSp.earned - takeFromEarned;
  1977→      if (restMu + restEarned <= 0) delete oS[colAsset]; else oS[colAsset] = { mu: restMu, earned: restEarned };
  1978→      const oB = { ...oPos.b };
  1979→      const newInt = accrued - interestPart;
  1980→      const newMu = bp.mu - principalPart;
  1981→      if (newMu + newInt <= 0) delete oB[asset]; else oB[asset] = { mu: newMu, int: newInt, atH: st.height };
  1982→      if (Object.keys(oS).length === 0 && Object.keys(oB).length === 0) delete positions[owner]; else positions[owner] = { s: oS, b: oB };
  1983→      st.lend = {
  1984→        ...ls,
  1985→        markets: { ...ls.markets, [asset]: { ...m, bMu: Math.max(0, m.bMu - principalPart), sEarnedMu: m.sEarnedMu + toSuppliers, resMu: m.resMu + toReserve, lastH: st.height }, [colAsset]: { ...ls.markets[colAsset]!, sMu: Math.max(0, ls.markets[colAsset]!.sMu - takeFromMu), sEarnedMu: Math.max(0, ls.markets[colAsset]!.sEarnedMu - takeFromEarned), lastH: st.height } },
  1986→        positions,
  1987→        totals: { ...ls.totals, borrowsMu: Math.max(0, ls.totals.borrowsMu - principalPart), interestPaidMu: ls.totals.interestPaidMu + interestPart, resToTreasuryMu: ls.totals.resToTreasuryMu + toReserve, liquidations: ls.totals.liquidations + 1 },
  1988→      };
  1989→      st.journal.push({ h: st.height, t: nowMs, soldier: issuer, action: "lend_liquidate", note: `liquidated ${owner}: paid ${pay}µ ${asset} · seized ${seizeMu}µ ${colAsset} (+5% bonus) · HF was ${lendHealthFactor(st, owner) / 100}` });
  1990→      return { ok: true, gas };
  1991→    }
  1992→    default:
  1993→      return { ok: false, err: "unhandled-op", gas: 0 };
  1994→  }
  1995→}
