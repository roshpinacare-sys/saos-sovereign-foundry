    25→function canon(v) {

    27→  if (Array.isArray(v)) return '[' + v.map(canon).join(',') + ']';

   132→/* 117p8-R8 — גובה-מזלג ענף-buy: מהבלוק-הראשון-מעל גובה-זה, מילוי-על-הזמנת-buy רץ
   133→ * בסמנטיקה-התקינה (taker-מוכר-SAOS). קבוע-חוזה-בקוד (לא-סביבה!) — אותו-ערך בכל מכונה.
   134→ * מתועד: spec §10 · worklog 117p8-R8. בלוקים ≤ גובה-זה משוחזרים byte-זהים להיסטוריה. */
   135→const FILL_BUY_FIX_HEIGHT = 130000;

   137→/* ── הפעלת טרנזאקציה בודדת (טהורה — ללא גישה לזמן/סביבה) ──
   138→ * atHeight (אופציונלי, 117p8-R8): גובה-הבלוק-שבו-ה-tx מיושם — נדרש לכללי-מזלג
   139→ * (fill-buy). buildBlock מעביר st.meta.height+1; replay-היסטוריה חייב להעביר
   140→ * את גובה-הבלוק-האמיתי. קריאה-2-ארגומנטים = ברירת-מחדל לשורה-הבאה. */
   141→function applyTx(st, tx, atHeight) {
   142→  if (!tx || typeof tx !== 'object') return { ok: false, gas: 0, err: 'bad-tx' };
   143→  const type = tx.type;
   144→  if (!GAS.hasOwnProperty(type)) return { ok: false, gas: 0, err: 'unknown-op' };
   145→  const gas = GAS[type];
   146→  /* 117p8-R8 — גובה-היישום (לכללי-המזלג): buildBlock מעביר גובה-הבלוק-החדש;
   147→   * replay-אמיתי מעביר גובה-הבלוק-ההיסטורי; קריאה-2-ארגומנטים = הבא-בתור. */
   148→  const atH = Number.isInteger(atHeight) ? atHeight : (st.meta.height || 0) + 1;
   149→  switch (type) {
   150→    case 'transfer':
   151→    case 'pay': {
   152→      const a = µ(Number(tx.amount));
   153→      if (!(a > 0)) return { ok: false, gas, err: 'amount' };
   154→      const from = acct(st, String(tx.from || '').toLowerCase());
   155→      const to = acct(st, String(tx.to || '').toLowerCase());
   156→      if (from.saos < a) return { ok: false, gas, err: 'insufficient' };
   157→      if (from === to && tx.from === tx.to) return { ok: false, gas, err: 'self' };
   158→      from.saos -= a; to.saos += a;
   159→      if (type === 'pay') st.meta.pays = (st.meta.pays || 0) + 1;
   160→      return { ok: true, gas };
   161→    }
   162→    case 'order': {
   163→      const owner = String(tx.owner || '').toLowerCase();
   164→      const side = tx.side === 'sell' ? 'sell' : tx.side === 'buy' ? 'buy' : null;
   165→      const a = µ(Number(tx.amount));
   166→      const p = Number(tx.price);
   167→      if (!side || !(a > 0) || !(p > 0)) return { ok: false, gas, err: 'bad-order' };
   168→      const o = acct(st, owner);
   169→      if (st.orders.filter((x) => x.owner === owner && x.live).length >= 8) return { ok: false, gas, err: 'order-cap' };
   170→      if (side === 'sell') {
   171→        if (o.saos < a) return { ok: false, gas, err: 'insufficient' };
   172→        o.saos -= a; o.lockedSaos += a;
   173→      } else {
   174→        const paySteem = Math.ceil(a * p);
   175→        if (o.steem < paySteem) return { ok: false, gas, err: 'insufficient-steem' };
   176→        o.steem -= paySteem; o.lockedSteem += paySteem;
   177→      }
   178→      const id = st.nextOrder++;
   179→      st.orders.push({ id, side, owner, amount: a, price: Math.round(p * 1e6), live: true, at: tx.at || null });
   180→      return { ok: true, gas, orderId: id };
   181→    }
   182→    case 'cancel': {
   183→      const id = Number(tx.orderId);
   184→      const ord = st.orders.find((x) => x.id === id && x.live);
   185→      if (!ord) return { ok: false, gas, err: 'no-order' };
   186→      if (ord.owner !== String(tx.owner || '').toLowerCase()) return { ok: false, gas, err: 'not-owner' };
   187→      ord.live = false;
   188→      const o = acct(st, ord.owner);
   189→      if (ord.side === 'sell') { o.lockedSaos -= ord.amount; o.saos += ord.amount; }
   190→      else { const back = Math.ceil(ord.amount * (ord.price / 1e6)); o.lockedSteem -= back; o.steem += back; }
   191→      return { ok: true, gas };
   192→    }
   193→    case 'fill': {
   194→      const taker = String(tx.taker || '').toLowerCase();
   195→      const id = Number(tx.orderId);
   196→      const ord = st.orders.find((x) => x.id === id && x.live);
   197→      if (!ord) return { ok: false, gas, err: 'no-order' };
   198→      if (ord.owner === taker) return { ok: false, gas, err: 'self-trade' };
   199→      let a = µ(Number(tx.amount));
   200→      if (!(a > 0)) return { ok: false, gas, err: 'amount' };
   201→      if (a > ord.amount) a = ord.amount;
   202→      const price = ord.price / 1e6;
   203→      const buyer = ord.side === 'sell' ? taker : ord.owner;
   204→      const seller = ord.side === 'sell' ? ord.owner : taker;
   205→      const b = acct(st, buyer), s = acct(st, seller);
   206→      /* 117p8-R8 — תיקון-מזלג ענף-buy (חוק-מספר, לא-סביבה):
   207→       * FILL_BUY_FIX_HEIGHT — מעליו ענף-buy תקין: ה-taker (מוכר-ה-SAOS) חייב-להחזיק
   208→       * SAOS ומוותר עליו; הבעלים מקבל SAOS ומשחרר-נעול עד-תקרת-המנעול (אף-יתרה-
   209→       * נעולה-שלילית). מתחתיו — הסמנטיקה-ההיסטורית byte-זהה (9,358 מילויים-חיים
   210→       * רצו-הפוך: הבעלים שילם SAOS+STEEM וה-taker קיבל את-שניהם — מתועד-ב-spec §10). */
   211→      if (ord.side === 'sell') {
   212→        // buyer משלם STEEM ישירות (יתרה-פנויה), seller ה-locked משוחרר ל-buyer-SAOS
   213→        const pay = Math.ceil(a * price);
   214→        if (b.steem < pay) return { ok: false, gas, err: 'insufficient-steem' };
   215→        b.steem -= pay; s.steem += pay;
   216→        ord.amount -= a; s.lockedSaos -= a; b.saos += a;
   217→      } else if (atH < FILL_BUY_FIX_HEIGHT) {
   218→        // buyer (ה-owner) כבר-נעל STEEM; taker מקבל את ה-STEEM ונותן SAOS מהפנוי
   219→        const pay = Math.ceil(a * price);
   220→        if (b.saos < a) return { ok: false, gas, err: 'insufficient' };
   221→        b.saos -= a; s.saos += a;
   222→        ord.amount -= a; b.lockedSteem -= pay; s.steem += pay;
   223→      } else {
   224→        // 117p8-R8 (fork @FILL_BUY_FIX_HEIGHT): taker-מוכר-SAOS · בעלים-מקבל ·
   225→        // שחרור-נעול מוגבל-במנעול-הקיים — אין-מנטרה, אין-שלילי, שימור-מלא.
   226→        const pay = Math.ceil(a * price);
   227→        if (s.saos < a) return { ok: false, gas, err: 'insufficient' };
   228→        const rel = Math.min(pay, Math.max(0, b.lockedSteem));
   229→        s.saos -= a; b.saos += a;
   230→        ord.amount -= a; b.lockedSteem -= rel; s.steem += rel;
   231→      }
   232→      if (ord.amount === 0) ord.live = false;
   233→      st.meta.trades = (st.meta.trades || 0) + 1;
   234→      st.meta.lastPrice = Math.round(price * 1e6);
   235→      return { ok: true, gas, filled: a };
   236→    }
   237→    case 'pool_add': {
   238→      const o = acct(st, String(tx.owner || '').toLowerCase());
   239→      const inS = µ(Number(tx.amountSaos)), inT = µ(Number(tx.amountSteem));
   240→      if (inS < 0 || inT < 0 || inS + inT <= 0) return { ok: false, gas, err: 'amount' };
   241→      if (o.saos < inS || o.steem < inT) return { ok: false, gas, err: 'insufficient' };
   242→      let shares;
   243→      if (st.pool.lpSupply <= 0) { shares = Math.round((inS + inT) * 1000); } // 1e3 µlp per µ
   244→      else { shares = Math.floor((inS / Math.max(st.pool.saos, 1)) * st.pool.lpSupply); if (!(shares > 0)) shares = 1; }
   245→      o.saos -= inS; o.steem -= inT;
   246→      st.pool.saos += inS; st.pool.steem += inT; st.pool.lpSupply += shares;
   247→      o.lp += shares;
   248→      return { ok: true, gas, shares };
   249→    }
   250→    case 'pool_swap': {
   251→      const t = String(tx.taker || '').toLowerCase();
   252→      const side = tx.side === 'sell_saos' ? 'sell_saos' : tx.side === 'buy_saos' ? 'buy_saos' : null;
   253→      const inAmt = µ(Number(tx.inAmt));
   254→      if (!side || !(inAmt > 0)) return { ok: false, gas, err: 'bad-swap' };
   255→      const o = acct(st, t);
   256→      const p = st.pool;
   257→      if (p.saos <= 0 || p.steem <= 0) return { ok: false, gas, err: 'empty-pool' };
   258→      if (side === 'sell_saos') {
   259→        if (o.saos < inAmt) return { ok: false, gas, err: 'insufficient' };
   260→        const fee = Math.floor(inAmt * p.feeBps / 10000);

   766→      return { ok: true, gas, purged: before - st.orders.length };
   767→    }
   768→    case 'produce': {
   769→      // STAGE-3 ROTATION (72): הפקה-במנדט — השרשרת עצמה אוכפת ראש-אחד-לסבב.
   770→      // החוק (§12): producer(round) = roster[round mod |roster|] — הרוסטר מפורסם.
   771→      // הקרנל אינו יודע את הרוסטר (נתון-סביבה) — הוא אוכף את המבנה ואת
   772→      // האכיפה-הקונסנסואלית: אין שני ראשים לאותו סבב (dup-round).
   773→      const p = String(tx.producer || '').toLowerCase();
   774→      const r = Number(tx.round);
   775→      const ref = String(tx.ref || '');
   776→      if (!p || !Number.isInteger(r) || r < 0 || !ref) return { ok: false, gas, err: 'bad-produce' };
   777→      st.producers = st.producers || {}; // יצירה-עצלנית — לא-נוגעים-במצב-לפני-הפקה-ראשונה
   778→      if (st.producers[r]) return { ok: false, gas, err: 'dup-round' };
   779→      st.producers[r] = { producer: p, ref };
   780→      for (const k of Object.keys(st.producers)) if (Number(k) < r - 31) delete st.producers[k];
   781→      st.meta.lastProducer = p;
   782→      st.meta.lastProduceRound = r;
   783→      return { ok: true, gas };
   784→    }
   785→    case 'relay': {
   786→      /* 108 — THE-DOOR (דבר-גורר-דבר): החוזה-החינמי שלנו מכתיב פעולה על רשת-מרוחקת.
   787→       * via: steem|blurt|tron · act: anchor|post|transfer|chain · seq מונוטוני פר-מנפיק.
   788→       * אפס-גז בכוונה — החינם הוא היתרון-המוחלט של הרשת-שלנו. מדיניות-מצב-כנה:
   789→       * open ≤ 48 (מניעת-נפיחה) · payload ≤ 400 תווים קנוניים · לוג-מסתובב 200.
   790→       * ה-relayer (חוץ-לקרנל) קורא queued → מבצע-באמת-על-הרשת-המרוחקת → relay_ack. */
   791→      const issuer = String(tx.issuer || '').toLowerCase();
   792→      const seq = Math.floor(Number(tx.seq));
   793→      const via = String(tx.via || '');
   794→      const act = String(tx.act || '');
   795→      const payload = tx.payload == null ? {} : tx.payload;
   796→      let payStr;
   797→      try { payStr = canon(payload); } catch (_) { return { ok: false, gas, err: 'relay-payload' }; }
   798→      if (!issuer) return { ok: false, gas, err: 'relay-issuer' };
   799→      if (!['steem', 'blurt', 'tron'].includes(via)) return { ok: false, gas, err: 'relay-via' };
   800→      if (!['anchor', 'post', 'transfer', 'chain'].includes(act)) return { ok: false, gas, err: 'relay-act' };
   801→      if (!Number.isInteger(seq) || seq < 1) return { ok: false, gas, err: 'relay-seq' };
   802→      if (payStr.length > 400) return { ok: false, gas, err: 'relay-payload-big' };
   803→      const R = st.relay = st.relay || { nextBy: {}, open: {}, log: [], sent: 0, issued: 0 };
   804→      const want = R.nextBy[issuer] || 1;
   805→      if (seq !== want) return { ok: false, gas, err: 'relay-seq-want-' + want };
   806→      if (Object.keys(R.open).length >= 48) return { ok: false, gas, err: 'relay-congested' };
   807→      const ref = issuer + ':' + seq;
   808→      /* 109 — ESCROW (החוזה-מחזיק-את-הכסף) — לדלתות-v:2 (שער-תאימות-לאחור):
   809→       * דלת-עם-budgetSun+ו:2 נועלת את-התקציב-כולו מאיזון-המנפיק ברגע-ההצתה.
   810→       * דלתות-109-המוקדמות (בלי-ו) חוזרות-ביתית byte-זהה לקרנל-שהנפיק-אותן —
   811→       * אפס-פיצול-שחזור (הלקח-מ-36666). מנפיק-בלי-כיסוי = דחייה-כנה.
   812→       * 111 — v:3 דלת-קציר: אותו-נעילה + עמלת-שירות feeBps (ברירת-מחדל 1000=10%,
   813→       * תקרה 2500) שנשארת-בחוזה במקום-פידיון-מלא — המרווח = קציר אמת
   814→       * מזרימת-אנרגיית-הטרנזקציות (v:2 נשאר חינם-מלא — החינם-הוא-היתרון).
   815→       * 111-fix (חוק-הברזל, נחשף-חי בסנדבוקס): כל-האימותים-והכיסוי קודמים
   816→       * למוטציה-הראשונה — דחייה-חייבת-להשאיר-את-המצב byte-זהה (לא-צורך-רצף,
   817→       * לא-דלת-נפתחה). נתיב-הצלחה byte-זהה לקרנל-109. */
   818→      const pv = Number(payload.v);
   819→      const budSun = Math.max(0, Math.min(1000000, Math.floor(Number(payload.budgetSun) || 0)));
   820→      let lockµ = 0, feeBps = 0;
   821→      if (pv === 2 || pv === 3) {
   822→        if (pv === 3 && !(budSun > 0)) return { ok: false, gas, err: 'relay-v3-budget' };
   823→        if (budSun > 0) {
   824→          const RATE = 1; // µSAOS ל-sun — תעריף-פרוטוקול-קבוע: 2000-sun דלת = 2 SAOS פיקדון (סדר-גודל-נמדד: יתרת-המנפיק)
   825→          lockµ = budSun * RATE;
   826→          const iss0 = acct(st, issuer);
   827→          if (iss0.saos < lockµ) return { ok: false, gas, err: 'relay-budget-unfunded(' + budSun + 'sun)' };
   828→          if (pv === 3) {
   829→            const fb = Number(payload.feeBps);
   830→            feeBps = Math.max(0, Math.min(2500, Number.isFinite(fb) ? Math.floor(fb) : 1000));
   831→          }
   832→        }
   833→      }
   834→      R.nextBy[issuer] = seq + 1;
   835→      R.open[ref] = { ref, via, act, payload: JSON.parse(payStr), by: issuer, status: 'queued', tries: 0 };
   836→      if (lockµ > 0) {
   837→        const iss = acct(st, issuer);
   838→        iss.saos -= lockµ; iss.lockedSaos += lockµ;
   839→        R.open[ref].escrowµ = lockµ;
   840→        R.open[ref].v = pv;
   841→        if (pv === 3) R.open[ref].feeBps = feeBps;
   842→      }
   843→      R.log.push(ref); if (R.log.length > 200) R.log.splice(0, R.log.length - 200);
   844→      R.issued++;
   845→      return { ok: true, gas, ref };
   846→    }
   847→    case 'relay_ack': {
   848→      /* 108 — הוכחת-ביצוע: רק מפעיל-הצומת (st.meta.issuer) רשאי לאשר — הדלת
   849→       * נסגרת ב-txid אמת מהרשת-המרוחקת. ack כפול נדחה: הוכחה-אחת לכל relay.
   850→       * 109 — GAS-SETTLE (שלם-להם-דרך-החוזה-שלנו): הפיקדון-כבר-ביד-החוזה
   851→       * (escrow מרגע-הצתת-הדלת). המבצע מוכיח spentSun; הוצאה-מעל-התקציב → דחייה.
   852→       * החוזה משלם-למבצע לפי-ההוצאה-האמת (תעריף-קבוע) ומחזיר-למנפיק את-העודף.
   853→       * Bandwidth-חינם = spentSun 0 = אפס-תשלום + פידיון-מלא — החינם-נשאר-היתרון. */
   854→      const opIssuer = String(st.meta.issuer || '').toLowerCase();
   855→      const acker = String(tx.issuer || '').toLowerCase();
   856→      const ref = String(tx.ref || '');
   857→      const txid = String(tx.txid || '').trim().slice(0, 96);
   858→      const spentSun = Math.max(0, Math.min(1000000, Math.floor(Number(tx.spentSun) || 0)));
   859→      if (!opIssuer || acker !== opIssuer) return { ok: false, gas, err: 'relay-ack-forbidden' };
   860→      const R = st.relay;
   861→      if (!R || !R.open[ref]) return { ok: false, gas, err: 'relay-unknown' };
   862→      const r = R.open[ref];
   863→      if (r.status !== 'queued') return { ok: false, gas, err: 'relay-acked' };
   864→      if (!txid) return { ok: false, gas, err: 'relay-txid' };
   865→      const budSun = Math.max(0, Math.min(1000000, Math.floor(Number((r.payload || {}).budgetSun) || 0)));
   866→      if (spentSun > budSun) return { ok: false, gas, err: 'spent-over-budget(' + spentSun + '>' + budSun + ')' };
   867→      if (r.escrowµ > 0) {
   868→        const RATE = 1; // אותו-תעריף-פרוטוקול — הפידיון-ההפוך-של-הנעילה
   869→        const payµ = Math.min(r.escrowµ, spentSun * RATE);
   870→        let feeµ = 0;
   871→        if (r.v === 3 && r.feeBps > 0) {
   872→          /* 111 — עמלת-הקציר: החוזה-מחזיק את-הסכום במקום-להחזיר.
   873→           * מוגבל-בשארית-הפידיון (לעולם-לא-נוגע בתשלום-המבצע). */
   874→          feeµ = Math.min(r.escrowµ - payµ, Math.floor(r.escrowµ * r.feeBps / 10000));
   875→        }
   876→        const iss = acct(st, r.by);
   877→        const exe = acct(st, acker);
   878→        iss.lockedSaos -= r.escrowµ;
   879→        iss.saos += r.escrowµ - payµ - feeµ; // עודף-חוזר-למנפיק (אחרי-עמלת-השירות)
   880→        exe.saos += payµ;                    // החוזה-משלם-למבצע
   881→        if (feeµ > 0) {
   882→          /* 111 — הכרה-אוטומטית-על-הספר: עמלת-הדלת = קציר-זרם-doors. */
   883→          const H = st.harvest = st.harvest || { potµ: 0, byStream: {}, seen: {}, seenRing: [], releases: [], count: 0, releasedµ: 0 };
   884→          const href = 'door:' + ref;
   885→          H.potµ += feeµ;
   886→          H.byStream.doors = (H.byStream.doors || 0) + feeµ;
   887→          if (!H.seen[href]) { H.seen[href] = 1; H.seenRing.push(href); if (H.seenRing.length > 512) { const old = H.seenRing.shift(); delete H.seen[old]; } H.count++; }
   888→          st.meta.harvestTotalµ = (st.meta.harvestTotalµ || 0) + feeµ;
   889→          r.harvestFeeµ = feeµ;
   890→        }
   891→        r.gasPaidµ = payµ;
   892→        r.refundµ = r.escrowµ - payµ - feeµ;
   893→        R.gasSettledSun = (R.gasSettledSun || 0) + spentSun;
   894→      }
   895→      r.status = 'sent'; r.txid = txid;
   896→      if (spentSun > 0) r.spentSun = spentSun;
   897→      R.sent++;
   898→      return { ok: true, gas, ref, txid, spentSun };
   899→    }
   900→    case 'harvest': {
   901→      /* 111 — THE-HARVEST (קציר-האנרגיה): הכרה-על-הספר בערך-שהפיקה-הרשת.
   902→       * השאלה-שהובילה-לכאן: ״יש טרנזקציה, יש אנרגיה שמתבזבזת, מישהו משלם —
   903→       * איך אנחנו מרוויחים משחרור-האנרגיה של הטרנזקציות?״
   904→       * התשובה: כל-זרימה-שעוברת-במסילות-שלנו משאירה-עמלה-אמת ביד-האוצר
   905→       * (עמלות-30bps של-הבריכות · הכנסות-הגריד · הזרמות-מסילה-חוצות · עמלות-דלתות-v:3).
   906→       * הפקודה מכירה-בערך-הזה על-הספר — מאגר-מדוד לפי-זרם, עם-ראיה (ref) אחת-לכל-ערך.
   907→       * רק-מפעיל-הצומת מכיר (מי-שמפעיל-את-המסילות-מודד-אותן) · dedupe-קשוח לפי-ref
   908→       * · תקרת-מאגר-seen (בקרת-נפיחה — אותה-משמעת כמו-טבעות-ה-weave). */
   909→      const ISSUER = st.meta.issuer || 'treasury';
   910→      if (String(tx.by || '').toLowerCase() !== ISSUER) return { ok: false, gas, err: 'harvest-forbidden' };
   911→      const stream = String(tx.stream || '');
   912→      if (['pools', 'grid', 'rail', 'doors'].indexOf(stream) < 0) return { ok: false, gas, err: 'harvest-stream' };
   913→      const amt = Math.max(0, Math.min(1e9, Math.floor(Number(tx.amountµ) || 0)));
   914→      if (!(amt > 0)) return { ok: false, gas, err: 'harvest-amount' };
   915→      const ref = String(tx.ref || '').trim().slice(0, 64);
   916→      if (!ref) return { ok: false, gas, err: 'harvest-ref' };
   917→      const H = st.harvest = st.harvest || { potµ: 0, byStream: {}, seen: {}, seenRing: [], releases: [], count: 0, releasedµ: 0 };
   918→      if (H.seen[ref]) return { ok: false, gas, err: 'harvest-dup' };
   919→      H.seen[ref] = 1; H.seenRing.push(ref);
   920→      if (H.seenRing.length > 512) { const old = H.seenRing.shift(); delete H.seen[old]; }
   921→      H.potµ += amt;
   922→      H.byStream[stream] = (H.byStream[stream] || 0) + amt;
   923→      H.count++;
   924→      st.meta.harvestTotalµ = (st.meta.harvestTotalµ || 0) + amt;
   925→      return { ok: true, gas, potµ: H.potµ };
   926→    }
   927→    case 'harvest_release': {
   928→      /* 111 — גלגל-החוזר: הפירות-נהנים. שחרור-ערך-מהמאגר לעבודת-הרשת:
   929→       * purpose: writer-reward (חיילים-שמאמתים/עוגנים) · gas-subsidy (מימון-תקציבי-דלת)
   930→       * · treasury-return. כפיפות-כלכלית-קשה: אי-אפשר-לשחרר-מעל-מה-שנקצר
   931→       * (potµ הוא-התקרה) — והכסף-האמת זז מיתרת-המפעיל אל-הנהנה.
   932→       * טבעת-שחרורים-100 — ההיסטוריה-המלאה-בפנקס-הבלוקים עצמו. */
   933→      const ISSUER = st.meta.issuer || 'treasury';
   934→      if (String(tx.by || '').toLowerCase() !== ISSUER) return { ok: false, gas, err: 'harvest-release-forbidden' };
   935→      const purpose = String(tx.purpose || '');
   936→      if (['writer-reward', 'gas-subsidy', 'treasury-return'].indexOf(purpose) < 0) return { ok: false, gas, err: 'harvest-purpose' };
   937→      const to = String(tx.to || '').toLowerCase();
   938→      if (!to) return { ok: false, gas, err: 'harvest-to' };
   939→      const amt = Math.max(0, Math.min(1e9, Math.floor(Number(tx.amountµ) || 0)));
   940→      if (!(amt > 0)) return { ok: false, gas, err: 'harvest-release-amount' };
   941→      const H = st.harvest;
   942→      if (!H || H.potµ < amt) return { ok: false, gas, err: 'harvest-pot-dry' };
   943→      const src = acct(st, ISSUER);
   944→      if (src.saos < amt) return { ok: false, gas, err: 'insufficient' };
   945→      const dst = acct(st, to);
   946→      src.saos -= amt; dst.saos += amt;
   947→      H.potµ -= amt;
   948→      H.releasedµ = (H.releasedµ || 0) + amt;
   949→      H.byPurpose = H.byPurpose || {};
   950→      H.byPurpose[purpose] = (H.byPurpose[purpose] || 0) + amt;
   951→      H.releases.push({ to, amountµ: amt, purpose, ref: String(tx.ref || '').slice(0, 64) || null, h: st.meta.height });
   952→      if (H.releases.length > 100) H.releases.shift();
   953→      return { ok: true, gas, potµ: H.potµ };
   954→    }
   955→    case 'sibling': {
   956→      /* 110 — אח-וריאפקה: רישום-אימות-עצמאי של ארטיפקטים-על-שרשרת-חוץ של-המופע-התאום.
   957→       * הסמנטיקה-הריבונית: הצומת-עצמו (st.meta.issuer) סרק את-השרשרת-הגולמית
   958→       * (STEEM get_account_history / trongrid) ומצא את-הראיה בעצמו — לא העתקת-טענות
   959→       * ממסך-של-אף-אחד. dedupe-קשוח לפי txid · מחסנית-מצב-עצלנית · מכסה-200.
   960→       * kind: door | step | ack | proof — מה-נבדק · verdict: verified | rejected. */
   961→      const ISSUER = st.meta.issuer || 'treasury';
   962→      if (String(tx.by || '').toLowerCase() !== ISSUER) return { ok: false, gas, err: 'sibling-forbidden' };
   963→      const stxid = String(tx.txid || '').trim().toLowerCase();
   964→      if (!/^[0-9a-f]{8,80}$/.test(stxid)) return { ok: false, gas, err: 'sibling-txid' };
   965→      const peer = String(tx.peer || '').trim().slice(0, 48);
   966→      const kind = ['door', 'step', 'ack', 'proof'].indexOf(String(tx.kind || '')) >= 0 ? String(tx.kind) : null;
   967→      if (!peer || !kind) return { ok: false, gas, err: 'sibling-shape' };
   968→      const S = st.siblings = st.siblings || { seen: {}, recs: [] };
   969→      if (S.seen[stxid]) return { ok: false, gas, err: 'sibling-dup' };
   970→      S.seen[stxid] = 1;
   971→      S.recs.push({
   972→        txid: stxid, peer, kind,
   973→        verdict: tx.verdict === 'rejected' ? 'rejected' : 'verified',
   974→        contract: String(tx.contract || '').slice(0, 64) || null,
   975→        step: Number.isInteger(Number(tx.step)) ? Number(tx.step) : null,
   976→        ref: String(tx.ref || '').slice(0, 64) || null,
   977→        note: String(tx.note || '').slice(0, 120) || null,
   978→        at: String(tx.at || '').slice(0, 32) || null,
   979→      });
   980→      if (S.recs.length > 200) { const dropped = S.recs.shift(); delete S.seen[dropped.txid]; }
   981→      st.meta.siblingsTotal = (st.meta.siblingsTotal || 0) + 1;
   982→      if (tx.verdict !== 'rejected') st.meta.siblingsVerified = (st.meta.siblingsVerified || 0) + 1;
   983→      return { ok: true, gas, txid: stxid };
   984→    }
   985→    case 'wallet': {
   986→      /* 113 — THE-WALLETS (ארנקי-הצבא): רישום-כתובות-רב-רשתי-מעוגן לכל-סוכן.
   987→       * הסמנטיקה-הריבונית: לכל-סוכן סט-ארנקים-פרטיים על-הרשתות-המובילות —
   988→       * evm (משרת ETH+BSC+Polygon+Arbitrum+Optimism+Base+Avalanche),
   989→       * tron, sol, btc (P2WPKH). digest = טביעת-המניפסט המעוגן-על-המארח
   990→       * (STEEM custom_json saos:wallets/v1 — ראיה-ציבורית-מתוארכת, אפס-עמלה).
   991→       * הכלל: המנפיק = headcorner (רישום-המניפסט) או הסוכן-עצמו (רישום-עצמי).
   992→       * מינימום-כתובת-אחת · אורכים-נבדקים · dedupe-קשוח (agent,digest) · מחסנית-200. */
   993→      const ISSUER = st.meta.issuer || 'treasury';
   994→      const by = String(tx.by || '').toLowerCase();
   995→      const agent = String(tx.agent || '').toLowerCase();
   996→      if (!/^[a-z0-9-]{3,20}$/.test(agent)) return { ok: false, gas, err: 'wallet-agent' };
   997→      if (by !== ISSUER && by !== agent) return { ok: false, gas, err: 'wallet-forbidden' };
   998→      const digest = String(tx.digest || '').trim().toLowerCase();
   999→      if (!/^[0-9a-f]{8,64}$/.test(digest)) return { ok: false, gas, err: 'wallet-digest' };
  1000→      const evm = String(tx.evm || '').trim().toLowerCase();
  1001→      const tron = String(tx.tron || '').trim();
  1002→      const sol = String(tx.sol || '').trim();
  1003→      const btc = String(tx.btc || '').trim();
  1004→      if (evm && !/^0x[0-9a-f]{40}$/.test(evm)) return { ok: false, gas, err: 'wallet-evm' };
  1005→      if (tron && !/^T[1-9A-HJ-NP-Za-km-z]{25,33}$/.test(tron)) return { ok: false, gas, err: 'wallet-tron' };
  1006→      if (sol && !/^[1-9A-HJ-NP-Za-km-z]{32,44}$/.test(sol)) return { ok: false, gas, err: 'wallet-sol' };
  1007→      if (btc && !/^bc1[a-z0-9]{11,71}$/.test(btc)) return { ok: false, gas, err: 'wallet-btc' };
  1008→      if (!evm && !tron && !sol && !btc) return { ok: false, gas, err: 'wallet-empty' };
  1009→      const hostTxid = String(tx.hostTxid || '').trim().toLowerCase();
  1010→      if (hostTxid && !/^[0-9a-f]{8,80}$/.test(hostTxid)) return { ok: false, gas, err: 'wallet-hosttxid' };
  1011→      const W = st.wallets = st.wallets || { seen: {}, recs: [] };
  1012→      const key = agent + '/' + digest;
  1013→      if (W.seen[key]) return { ok: false, gas, err: 'wallet-dup' };
  1014→      W.seen[key] = 1;
  1015→      W.recs.push({
  1016→        agent, digest,
  1017→        evm: evm || null, tron: tron || null, sol: sol || null, btc: btc || null,
  1018→        hostTxid: hostTxid || null,
  1019→        by, selfReg: by === agent,
  1020→        at: String(tx.at || '').slice(0, 32) || null,
  1021→      });
  1022→      if (W.recs.length > 200) { const dropped = W.recs.shift(); delete W.seen[dropped.agent + '/' + dropped.digest]; }
  1023→      st.meta.walletsTotal = (st.meta.walletsTotal || 0) + 1;
  1024→      st.meta.walletsAgents = st.wallets.recs.length;
  1025→      return { ok: true, gas, agent };
  1026→    }
  1027→    default:
  1028→      return { ok: false, gas, err: 'unknown-op' };
  1029→  }
  1030→}

  1033→function buildBlock(st, txs, time) {
  1034→  const results = [];
  1035→  for (const tx of txs) {
  1036→    const r = applyTx(st, tx, (st.meta.height || 0) + 1);
  1037→    results.push({ ok: r.ok, err: r.err || null, gas: r.gas, filled: r.filled || undefined, out: r.out || undefined, shares: r.shares || undefined, orderId: r.orderId || undefined, type: tx.type });
  1038→  }
  1039→  st.meta.height = (st.meta.height || 0) + 1;
  1040→  st.meta.lastTime = time;
  1041→  const block = { height: st.meta.height, time, prevHash: st.meta.lastBlockHash || null, txs, results, gasUsed: results.reduce((s, r) => s + (r.gas || 0), 0) };
  1042→  const blockHash = sha256(canon(block));
  1043→  block.hash = blockHash;
  1044→  st.meta.lastBlockHash = blockHash;
  1045→  return { block, results };
  1046→}
