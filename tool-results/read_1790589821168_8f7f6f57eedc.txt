   100→const IFACES = new Map<string, ethers.Interface>();
   101→function ifaceOf(name: string): ethers.Interface | null {
   102→  let f = IFACES.get(name);
   103→  if (!f) {
   104→    const c = ARTIFACT.contracts.find((x) => x.name === name);
   105→    if (!c) return null;
   106→    f = new ethers.Interface(c.abi);
   107→    IFACES.set(name, f);
   108→  }
   109→  return f;
   110→}

   164→// ─── event decoding ───────────────────────────────────────────────────────

   166→interface DecodedEvent {
   167→  contract: string;
   168→  name: string;
   169→  values: Record<string, string>;
   170→  block: number;
   171→  txid: string;
   172→}
   173→
   174→function decodeLog(contractName: string, log: { address: string; topics: string[]; data: string }): DecodedEvent | null {
   175→  const iface = ifaceOf(contractName);
   176→  if (!iface) return null;
   177→  try {
   178→    const parsed = iface.parseLog({ topics: log.topics, data: log.data });
   179→    if (!parsed) return null;
   180→    const values: Record<string, string> = {};
   181→    parsed.fragment.inputs.forEach((inp, i) => {
   182→      values[inp.name] = argStr(parsed.args[i]);
   183→    });
   184→    return { contract: contractName, name: parsed.name, values, block: -1, txid: "" };
   185→  } catch {
   186→    return null;
   187→  }
   188→}
   189→
   190→function decodeLogs(raw: Array<{ address: string; topics: string[]; data: string; blockNumber: number; transactionHash: string }>, state: DeploymentState): DecodedEvent[] {
   191→  const byAddr = new Map<string, string>();
   192→  for (const [name, addr] of Object.entries(state.contracts)) byAddr.set(addr.toLowerCase(), name);
   193→  const out: DecodedEvent[] = [];
   194→  for (const log of raw) {
   195→    const cname = byAddr.get(log.address.toLowerCase());
   196→    if (!cname) continue;
   197→    const ev = decodeLog(cname, log);
   198→    if (ev) {
   199→      ev.block = log.blockNumber;
   200→      ev.txid = log.transactionHash;
   201→      out.push(ev);
   202→    }
   203→  }
   204→  return out;
   205→}

   207→// ─── run assembly ─────────────────────────────────────────────────────────
   208→
   209→function buildRun(
   210→  id: number,
   211→  runRow: { initiator: string; intent: string; startedAt: number; completedAt: number; status: number },
   212→  events: DecodedEvent[],
   213→  startGas: bigint | null
   214→): MeshRunView {
   215→  const mine = events.filter((e) => {
   216→    const rid = e.values["runId"] ?? e.values["id"];
   217→    return rid === String(id);
   218→  });
   219→
   220→  const dispatched = mine.find((e) => e.name === "RunDispatched");
   221→  const completed = mine.find((e) => e.name === "RunCompleted");
   222→  const scored = mine.find((e) => e.name === "ValueScored");
   223→  const reward = mine.find((e) => e.name === "RewardPaid");
   224→  const solution = mine.find((e) => e.name === "SolutionCommitted");
   225→  const hopEvents = mine
   226→    .filter((e) => e.name === "Hop")
   227→    .sort((a, b) => Number(a.values["index"] ?? 0) - Number(b.values["index"] ?? 0));
   228→
   229→  const gasUsed = completed ? Number(completed.values["gasUsed"] ?? 0) : scored ? Number(scored.values["gasUsed"] ?? 0) : null;
   230→
   231→  // hop gas: gasBefore[i] − gasBefore[i+1]; the last hop closes on the
   232→  // exhaustion point (startGas − gasUsed) — honest gas trail per hop.
   233→  const hopTrace: MeshRunHop[] = hopEvents.map((h, i) => {
   234→    const gb = BigInt(h.values["gasBefore"] ?? "0");
   235→    let gas: number | null = null;
   236→    if (i + 1 < hopEvents.length) {
   237→      gas = Number(gb - BigInt(hopEvents[i + 1]!.values["gasBefore"] ?? "0"));
   238→    } else if (startGas !== null && gasUsed !== null) {
   239→      gas = Math.max(0, Number(gb - (startGas - BigInt(gasUsed))));
   240→    }
   241→    const link = h.values["link"] ?? "";
   242→    return {
   243→      index: Number(h.values["index"] ?? i),
   244→      link: state7Name(link),
   245→      role: MESH_ROLES[state7Name(link)] ?? "",
   246→      address: link,
   247→      gas
   248→    };
   249→  });
   250→
   251→  const anchorEvents = mine.filter((e) => e.name === "AnchorStamped");
   252→  const orderDelivered = mine.find((e) => e.name === "OrderDelivered");
   253→
   254→  return {
   255→    id,
   256→    intent: runRow.intent,
   257→    initiator: runRow.initiator,
   258→    status: runRow.status === 1 ? "completed" : "open",
   259→    startedAt: Number(runRow.startedAt) * 1000,
   260→    completedAt: Number(runRow.completedAt) > 0 ? Number(runRow.completedAt) * 1000 : null,
   261→    durationMs:
   262→      Number(runRow.completedAt) > 0 && Number(runRow.startedAt) > 0
   263→        ? (Number(runRow.completedAt) - Number(runRow.startedAt)) * 1000
   264→        : null,
   265→    gasUsed,
   266→    score: scored ? Number(scored.values["score"] ?? 0) : null,
   267→    hops: scored ? Number(scored.values["hops"] ?? 0) : null,
   268→    rewardTo: reward?.values["to"] ?? null,
   269→    rewardAmount: reward ? ethers.formatEther(BigInt(reward.values["amount"] ?? "0")) : null,
   270→    solutionTitle: solution?.values["title"] ?? null,
   271→    solutionDigest: solution?.values["digest"] ?? null,
   272→    order: orderDelivered
   273→      ? {
   274→          id: Number(orderDelivered.values["orderId"] ?? 0),
   275→          mission: orderDelivered.values["mission"] ?? "",
   276→          soldier: orderDelivered.values["soldier"] ?? "",
   277→          reward: ethers.formatEther(BigInt(orderDelivered.values["reward"] ?? "0")),
   278→          paid: (orderDelivered.values["paid"] ?? "false") === "true"
   279→        }
   280→      : null,
   281→    anchors: anchorEvents.map((a) => ({
   282→      chain: a.values["chain"] ?? "",
   283→      height: a.values["height"] ?? "0",
   284→      digest: a.values["digest"] ?? ""
   285→    })),
   286→    hopTrace,
   287→    txid: dispatched?.txid ?? mine[0]?.txid ?? "",
   288→    block: dispatched?.block ?? mine[0]?.block ?? 0
   289→  };
   290→}

   460→export async function meshFeed(limit = 8): Promise<MeshFeedData> {
   461→  const state = readState();
   462→  if (!state?.ok) return { ok: false, runs: [], transfers: [], agents: [], rails: [], orders: [], error: "SAOS-EVM לא פרוס" };
   463→  try {
   464→    const provider = getProvider();
   465→    const hub = contract("SAOSMeshHub", state);
   466→    const relay = contract("SAOSMeshRelay", state);
   467→    const ledger = contract("SAOSLedger", state);
   468→    const bridge = contract("SAOSBridge", state);
   469→    if (!hub || !relay || !ledger || !bridge) throw new Error("artifact/state mismatch");
   470→
   471→    const addresses = Object.values(state.contracts);
   472→    const [logs, runCount] = await Promise.all([
   473→      provider.getLogs({ address: addresses, fromBlock: state.deployBlock }),
   474→      hub.runCount()
   475→    ]);
   476→    const events = decodeLogs(
   477→      logs.map((l) => ({
   478→        address: l.address,
   479→        topics: [...l.topics],
   480→        data: l.data,
   481→        blockNumber: l.blockNumber,
   482→        transactionHash: l.transactionHash
   483→      })),
   484→      state
   485→    );
   486→
   487→    // ── recent runs (storage truth + event trail)
   488→    const runs: MeshRunView[] = [];
   489→    const n = Number(runCount);
   490→    for (let id = n; id >= Math.max(1, n - limit + 1); id--) {

   623→export async function runMesh(intentRaw: string, opts?: { soldier?: boolean }): Promise<MeshRunOutcome> {
   624→  const state = readState();
   625→  if (!state?.ok) return { ok: false, run: null, anchorsStamped: 0, error: "SAOS-EVM לא פרוס" };
   626→  const intent = (intentRaw ?? "").trim().slice(0, 120) || "mesh run: manual dispatch";
   627→
   628→  try {
   629→    const provider = getProvider();
   630→    // v17 — work dispatches rotate REAL soldier signers (1..5); plain
   631→    // mesh runs keep the operator (signer 0). All anvil unlocked accounts.
   632→    let runner: ethers.Signer;
   633→    if (opts?.soldier) {
   634→      const hub0 = contract("SAOSMeshHub", state);
   635→      const n = hub0 ? Number(await hub0.runCount()) : 0;
   636→      runner = await provider.getSigner(1 + (n % 5));
   637→    } else {
   638→      runner = await provider.getSigner(0);
   639→    }
   640→    const hub = contract("SAOSMeshHub", state, runner);
   641→    const bridge = contract("SAOSBridge", state, runner);
   642→    if (!hub || !bridge) throw new Error("artifact/state mismatch");
   643→
   644→    // 1. LIVE anchors flow INTO the mesh right before the run (דבר מוביל לדבר)
   645→    let anchorsStamped = 0;
   646→    const heights: Array<{ chain: string; height: number }> = [];
   647→    try {
   648→      const lastSeal = await db.sealRecord.findFirst({ orderBy: { height: "desc" }, select: { height: true } });
   649→      if (lastSeal) heights.push({ chain: "SAOS-NET", height: lastSeal.height });
   650→      const [eth, bsc, sol] = await Promise.all([probeEthereum(), probeBsc(), probeSolana()]);
   651→      if (eth.ok && eth.blockNumber) heights.push({ chain: "ETHEREUM", height: eth.blockNumber });
   652→      if (bsc.ok && bsc.blockNumber) heights.push({ chain: "BSC", height: bsc.blockNumber });
   653→      if (sol.ok && sol.slot) heights.push({ chain: "SOLANA", height: sol.slot });
   654→    } catch {
   655→      /* anchors best-effort — the run still happens */
   656→    }
   657→    if (heights.length > 0) {
   658→      try {
   659→        const tx = await bridge.anchorExternalBulk(
   660→          heights.map((h) => h.chain),
   661→          heights.map((h) => BigInt(h.height)),
   662→          heights.map((h) => ethers.id(`saos:${h.chain}:${h.height}`))
   663→        );
   664→        const rc = await waitTx(tx); // v57 — שער שקט נסגר; העוגנים אידמפוטנטיים ויטבעו בריצה הבאה
   665→        if (rc) anchorsStamped = heights.length;
   666→      } catch {
   667→        /* honest — anchors are best-effort */
   668→      }
   669→    }
   670→
   671→    // 2. dispatch through the whole chain
   672→    const tx = await hub.dispatch(intent);
   673→    const rc = await waitTx(tx); // v57 — null = שער-המתנה שקט נסגר (לא revert)
   674→    if (rc === null) throw new Error("קבלת העסקה לא אושרה בחלון 15ש' — העסקה עשויה להיכרות; אמת מהשרשרת בקריאה הבאה (honest)");
   675→    if (rc.status !== 1) throw new Error("mesh run reverted (honest)");
   676→
   677→    // 3. build the trace from the receipt logs + storage
   678→    const events = decodeLogs(
   679→      rc.logs.map((l) => ({
   680→        address: l.address,
   681→        topics: [...l.topics],
   682→        data: l.data,
   683→        blockNumber: l.blockNumber,
   684→        transactionHash: l.transactionHash
   685→      })),
   686→      state
   687→    );
   688→    const runId = Number(events.find((e) => e.name === "RunDispatched")?.values["runId"] ?? 0);
   689→    const row = (await hub.runs(runId)) as unknown as [string, string, bigint, bigint, number];
   690→    const startGas = (await contract("SAOSMeshRelay", state)!.startGas(runId)) as bigint;
   691→    const run = buildRun(
   692→      runId,
   693→      { initiator: String(row[0]), intent: String(row[1]), startedAt: Number(row[2]), completedAt: Number(row[3]), status: Number(row[4]) },
   694→      events,
   695→      startGas
   696→    );
   697→
   698→    return { ok: true, run, anchorsStamped };
   699→  } catch (e) {
   700→    return { ok: false, run: null, anchorsStamped: 0, error: e instanceof Error ? e.message.slice(0, 160) : "mesh run failed" };
   701→  }
   702→}

   919→export async function anchorWorkDigest(
   920→  orderId: number,
   921→  digest: string
   922→): Promise<{ ok: boolean; txid?: string; block?: number; error?: string }> {
   923→  const state = readState();
   924→  if (!state?.ok) return { ok: false, error: "SAOS-EVM לא פרוס" };
   925→  const bridgeAddr = state.contracts["SAOSBridge"];

   930→    const bridge = contract("SAOSBridge", state, hq);
   931→    if (!bridge) throw new Error("artifact/state mismatch");

   933→    try {
   934→      await waitTx(await bridge.registerChain(WORK_RAIL)); // v57 — דדליין 15ש'
   935→    } catch {
   936→      /* rail already registered — pass */
   937→    }
   938→    const tx = await bridge.anchorExternal(WORK_RAIL, BigInt(orderId), digest);
   939→    const rc = await waitTx(tx); // v57 — null = שער-המתנה שקט נסגר (לא revert)
   940→    if (rc === null) return { ok: false, error: "קבלת העיגון לא אושרה בחלון 15ש' — העיגון עשוי להיכרות; אמת מהשרשרת בקריאה הבאה (honest)" };
   941→    if (rc.status !== 1) throw new Error("anchorExternal reverted (honest)");
   942→    return { ok: true, txid: rc.hash, block: rc.blockNumber ?? 0 };
   943→  } catch (e) {
   944→    return { ok: false, error: e instanceof Error ? e.message.slice(0, 160) : "work anchor failed" };
   945→  }
   946→}

   948→/** live state of the SAOS-WORK rail (height = last delivered orderId). */
   949→export async function workRail(): Promise<{ chain: string; height: string; digest: string; at: number } | null> {
   950→  const state = readState();
   951→  if (!state?.ok) return null;
   952→  try {
   953→    const bridge = contract("SAOSBridge", state);
   954→    if (!bridge) return null;
   955→    const a = (await bridge.latestAnchor(WORK_RAIL)) as unknown as [string, bigint, string, bigint];
   956→    return { chain: String(a[0]), height: String(a[1]), digest: String(a[2]), at: Number(a[3]) * 1000 };
   957→  } catch {
   958→    return null;
   959→  }
   960→}
