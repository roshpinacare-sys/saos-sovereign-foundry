   116→export async function fetchJson(url: string, timeoutMs = 4000): Promise<Record<string, unknown> | null> {
   117→  try {
   118→    // v24 מגן (runner): the mesh service talks ONLY to the local platform
   119→    // (127.0.0.1) — any other host is blocked before connect (fail-closed,
   120→    // anti-phishing: unknown hosts are never probed from the runner).
   121→    const u = new URL(url);
   122→    if (u.hostname !== "127.0.0.1" && u.hostname !== "localhost") {
   123→      console.error(`[magen] blocked outbound fetch to ${u.hostname} (loopback only)`);
   124→      return null;
   125→    }
   126→    // v55 לולאת-המוות שנתפסה חי (restart #1/#2): בזמן מצוקת-ווב (RSS 1.5GB ·
   127→    // page-hang) הווב מחזיר 200+כותרות אבל הגוף נתקע — ו-bun לעולם לא דוחה
   128→    // את res.json() ב-abort (הוכח בריפרו repro-abort.ts ניסוי B: תלייה>15ש'
   129→    // למרות abort). הטיפול: ה-abort ממשיך לכסות גם את קריאת-הגוף (clearTimeout
   130→    // רק ב-finally), וקריאת-הגוף עצמה רצה במירוץ קשיח נגד deadline שתמיד
   131→    // מסתיים — לא סומכים על ה-abort, ה-fetch-כולו חסום-זמן.
   132→    const ctl = new AbortController();
   133→    const t = setTimeout(() => ctl.abort(), timeoutMs);
   134→    try {
   135→      const res = await fetch(url, { signal: ctl.signal });
   136→      if (!res.ok) return null;
   137→      // v55 — המירוץ הקשיח עם ניקוי-טיימר: המנצח מנקה את ה-deadline (אין
   138→      // לוגי-שקר מאוחרים); המפסיד מוחזר כ-null כנה.
   139→      const body = await new Promise<Record<string, unknown> | null>((resolve) => {
   140→        const t2 = setTimeout(() => {
   141→          console.log(`[evm-mesh] v55 body-deadline: ${u.pathname} stalled >${timeoutMs}ms — honest null (bun abort does not reject a stalled body read)`);
   142→          resolve(null);
   143→        }, timeoutMs);
   144→        res.json().then(
   145→          (v) => { clearTimeout(t2); resolve(v as Record<string, unknown>); },
   146→          () => { clearTimeout(t2); resolve(null); }
   147→        );
   148→      });
   149→      return body;
   150→    } finally {
   151→      clearTimeout(t);
   152→    }
   153→  } catch {
   154→    return null;
   155→  }
   156→}

   158→/** LIVE anchors from the platform's 8-chain grid (best-effort). */
   159→export async function fetchLiveAnchors(): Promise<{
   160→  chains: string[];
   161→  heights: bigint[];
   162→  digests: string[];
   163→} | null> {
   164→  const [probes, status] = await Promise.all([
   165→    fetchJson("http://127.0.0.1:3000/api/saos/live-probes"),
   166→    fetchJson("http://127.0.0.1:3000/api/saos/status")
   167→  ]);
   168→  const h: Record<string, number> = {};
   169→  const stats = status?.stats as { lastHeight?: number } | undefined;
   170→  if (stats?.lastHeight != null) h["SAOS-NET"] = Number(stats.lastHeight);
   171→  const eth = probes?.eth as { blockNumber?: number } | undefined;
   172→  const bsc = probes?.bsc as { blockNumber?: number } | undefined;
   173→  const sol = probes?.sol as { slot?: number } | undefined;
   174→  if (eth?.blockNumber != null) h["ETHEREUM"] = Number(eth.blockNumber);
   175→  if (bsc?.blockNumber != null) h["BSC"] = Number(bsc.blockNumber);
   176→  if (sol?.slot != null) h["SOLANA"] = Number(sol.slot);
   177→
   178→  const chains: string[] = [];
   179→  const heights: bigint[] = [];
   180→  const digests: string[] = [];
   181→  for (const c of MESH_CHAINS) {
   182→    const v = h[c];
   183→    if (v != null && Number.isFinite(v) && v > 0) {
   184→      chains.push(c);
   185→      heights.push(BigInt(v));
   186→      digests.push(ethers.id(`saos:${c}:${v}`));
   187→    }
   188→  }
   189→  return chains.length > 0 ? { chains, heights, digests } : null;
   190→}

   225→export async function fetchRealAssets(): Promise<{ assets: Record<string, RealAsset>; live: number }> {
   226→  const flow = await fetchJson("http://127.0.0.1:3000/api/saos/flow", 25000);
   227→  const rows = (((flow?.radar as { assets?: unknown[] } | undefined)?.assets) ?? []) as Array<{
   228→    symbol?: string;
   229→    holder?: string;
   230→    address?: string;
   231→    balanceNum?: number;
   232→    priceUsd?: number | null;
   233→  }>;
   234→  const pick = (symbol: string, prefix: string): RealAsset | null => {
   235→    const a = rows.find(
   236→      (x) => x.symbol === symbol && x.holder === "headcorner" && typeof x.address === "string" && x.address.startsWith(prefix)
   237→    );
   238→    if (!a || typeof a.balanceNum !== "number" || !(a.balanceNum > 0)) return null;
   239→    if (typeof a.priceUsd !== "number" || !(a.priceUsd > 0)) return null;
   240→    return {
   241→      symbol,
   242→      units: a.balanceNum,
   243→      usd: a.priceUsd,
   244→      source: `מכ"ם חי ${new Date().toISOString().slice(0, 10)}`,
   245→      address: a.address as string
   246→    };
   247→  };
   248→  const out: Record<string, RealAsset> = {};
   249→  let live = 0;
   250→  for (const [sym, prefix] of [
   251→    ["TRX", "TG6Srs"],
   252→    ["SOL", "9knfnF"],
   253→    ["ETH", "0x2767"]
   254→  ] as const) {
   255→    const got = pick(sym, prefix);
   256→    if (got) {
   257→      out[sym] = got;
   258→      live += 1;
   259→    } else {
   260→      const fb = MEASURED_SNAPSHOT[sym]!;

   263→  }
   264→  return { assets: out, live };

   283→async function sendTx(label: string, call: () => Promise<ethers.ContractTransactionResponse>): Promise<ethers.ContractTransactionReceipt | null> {
   284→  const tx = await call();
   285→  // v91 receipt-poll fix (measured 2026-09-13): tx.wait() hung forever even
   286→  // though the tx DID mine (block 11 carried the hub.wire selector, zero "✓"
   287→  // lines, 1019 blockNumber heartbeats, empty txpool) — the ethers polling
   288→  // transaction-subscriber race from the v61 Promise-Guard family. We no
   289→  // longer blindly trust wait(): poll the chain for the receipt ourselves
   290→  // and fail LOUDLY on timeout so the watchdog loop recovers honestly.
   291→  let rc: ethers.ContractTransactionReceipt | null = null;
   292→  for (let i = 0; i < 240 && !rc; i++) {
   293→    try {
   294→      rc = (await tx.provider.getTransactionReceipt(tx.hash)) as ethers.ContractTransactionReceipt | null;
   295→    } catch {
   296→      // transient RPC hiccup — keep polling, the chain is the truth
   297→    }
   298→    if (!rc) await new Promise((r) => setTimeout(r, 250));
   299→  }
   300→  if (!rc) {
   301→    console.error(`[mesh] sendTx ${label}: no receipt in 60s for ${tx.hash} — fail-loud (watchdog recovers)`);
   302→    throw new Error(`sendTx ${label}: receipt timeout for ${tx.hash}`);
   303→  }
   304→  console.log(`[mesh] ✓ ${label} (block ${rc.blockNumber})`);
   305→  return rc;
   306→}

   320→export async function deployMesh(): Promise<DeploymentState> {
   321→  const artifact = loadArtifact();
   322→  fs.mkdirSync(path.dirname(STATE_PATH), { recursive: true });
   323→
   324→  // v54 מכוון-הרשת — ריעוף 120ms לפריסה/המתנות
   325→  const provider = new ethers.JsonRpcProvider(MESH_RPC, MESH_CHAIN_ID, { batchMaxCount: 1, pollingInterval: 120 });
   326→  const operator = await provider.getSigner(0);
   327→  const operatorAddr = await operator.getAddress();
   328→
   329→  const abiOf = (name: string): ethers.InterfaceAbi => {
   330→    const c = artifact.contracts.find((x) => x.name === name);
   331→    if (!c) throw new Error(`artifact missing ${name}`);
   332→    return c.abi;
   333→  };
   334→
   335→  // 1. deploy all seven contracts
   336→  const addr: Record<string, string> = {};
   337→  for (const c of artifact.contracts) {
   338→    const factory = new ethers.ContractFactory(c.abi, c.bytecode, operator);
   339→    const deployed = await factory.deploy();
   340→    await deployed.waitForDeployment();
   341→    addr[c.name] = await deployed.getAddress();
   342→    console.log(`[mesh] deployed ${c.name} → ${addr[c.name]} (${c.bytecodeBytes}B)`);
   343→  }
   344→
   345→  const C = (name: string) => new ethers.Contract(addr[name] as string, abiOf(name), operator);
   346→  const hub = C("SAOSMeshHub") as unknown as HubContract;
   347→  const relay = C("SAOSMeshRelay");
   348→  const identity = C("SAOSIdentity");
   349→  const ledger = C("SAOSLedger") as unknown as LedgerContract;
   350→  const solutions = C("SAOSSolutions");
   351→  const bridge = C("SAOSBridge");
   352→  const workboard = C("SAOSWorkBoard");
   353→  const microgrid = C("SAOSMicrogrid");
   354→  const exchange = C("SAOSExchange") as unknown as ExchangeContract;
   355→  const value = C("SAOSValueEngine");
   356→
   357→  // v4 — אוצר-הבורסה / reserve custodian: חותם 9 (anvil unlocked). כל הקציר
   358→  //      וההון-יוצר-השוק של הבורסה יושבים כאן — כתובת פומבית בלבד, אין מפתח.
   359→  const treasurySigner = await provider.getSigner(9);
   360→  const treasuryAddr = await treasurySigner.getAddress();
   361→
   362→  // 2. wire the chain — every arrow below is a real contract-to-contract link
   363→  await sendTx("hub → relay + valueEngine", () => hub.wire(addr.SAOSMeshRelay, addr.SAOSValueEngine));
   364→  await sendTx("relay → hub", () => relay.wireHub(addr.SAOSMeshHub));
   365→  for (const link of LINK_ORDER) {
   366→    await sendTx(`relay link + ${link}`, () => relay.addLink(addr[link] as string));
   367→  }
   368→  await sendTx("identity → relay", () => identity.wire(addr.SAOSMeshRelay as string));
   369→  await sendTx("ledger → relay", () => ledger.wire(addr.SAOSMeshRelay as string));
   370→  await sendTx("solutions → relay", () => solutions.wire(addr.SAOSMeshRelay as string));
   371→  await sendTx("bridge → relay", () => bridge.wire(addr.SAOSMeshRelay as string));
   372→  await sendTx("workboard → relay + ledger", () => workboard.wire(addr.SAOSMeshRelay as string, addr.SAOSLedger as string));
